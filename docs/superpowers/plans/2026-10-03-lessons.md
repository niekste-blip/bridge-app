# Leerpad: lessen 1 tot en met 7 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Het leerpad in de app: lessen met uitleg en oefeningen, voortgang die lessen openzet, een oefenronde voor wat fout ging, en de inhoud van les 1 tot en met 7.

**Architecture:** De module `lessons` bevat alles zonder schermcode. Een les is een JSON-bestand met pagina's uitleg en oefeningen van drie soorten: een meerkeuzevraag, een biedvraag en een speelopdracht. Het goede antwoord op een biedvraag staat niet in het bestand: het is wat de regeltabel uit `bidding` voorschrijft voor die les. Een speelopdracht is een kleine stand die de rekenmodule volledig kan doorrekenen. Een test controleert elke oefening tegen de regeltabel en de rekenmodule. De app krijgt een leerpadscherm en een lesscherm; de voortgang wordt bewaard en bepaalt wat de bots mogen bieden.

**Tech Stack:** TypeScript (strict), React 19, Vitest. Bouwt op alle eerdere modules.

Spec: `docs/superpowers/specs/2026-10-03-bridge-app-design.md`, stap 7 van de bouwvolgorde. Vereist dat het plan `2026-10-03-coach.md` is uitgevoerd.

## Afspraken over het gedrag

- Een les heeft drie tot vijf pagina's uitleg en vijf tot tien oefeningen.
- Bij een fout verschijnt meteen de uitleg, en de oefening komt aan het eind van dezelfde les terug.
- Een les is afgerond als minstens 80 procent van de oefeningen in één keer goed was. Dan gaat de volgende les open.
- Oefeningen die in één keer fout gingen komen op de lijst "Oefen wat ik lastig vond". Een oefening verdwijnt daar na twee keer achter elkaar goed.
- In een speelopdracht is de leerling leider in Zuid met Noord als dummy. Oost en West worden gespeeld door de bot op het hoogste niveau. De opdracht is gehaald als Noord-Zuid het gevraagde aantal slagen maakt; dat aantal is precies wat tegen foutloos tegenspel haalbaar is.
- In vrij spelen biedt de leerling pas zelf als les 4 is afgerond. Tot die tijd bieden de bots en speelt de leerling het contract.
- De bots gebruiken biedregels tot en met de hoogst afgeronde les, met les 7 als ondergrens.
- De instelling "Alle lessen openzetten" zet alles open en laat de bots alle regels gebruiken, voor wie al kan bridgen (en voor het uitproberen van de app).
- De lesteksten spreken de leerling aan met u.

```ts
import { SEATS, STRAINS, type Seat, type Strain } from '../core';

// A question with a few answers to choose from
export interface ChoiceExercise {
  readonly type: 'choice';
  readonly id: string;
  readonly question: string;
  // A hand to show with the question, as spades.hearts.diamonds.clubs
  readonly hand?: string;
  readonly options: readonly string[];
  // Index of the right option
  readonly answer: number;
  readonly explanation: string;
}

// A hand and an auction; the right call is what the bidding rules of the lesson prescribe
export interface BidExercise {
  readonly type: 'bid';
  readonly id: string;
  readonly hand: string;
  readonly dealer: Seat;
  // The calls so far, for example '1H P'; the student is the next to call
  readonly auction: string;
}

// A small position that South plays as declarer with North as dummy
export interface PlayExercise {
  readonly type: 'play';
  readonly id: string;
  readonly intro: string;
  readonly trump: Strain;
  readonly leader: Seat;
  // Cards per seat, for example 'SA SK H3'
  readonly hands: Readonly<Record<Seat, string>>;
  // Tricks North-South must take
  readonly target: number;
  readonly explanation: string;
}

export type Exercise = ChoiceExercise | BidExercise | PlayExercise;

export interface Page {
  readonly title: string;
  // Paragraphs separated by an empty line
  readonly text: string;
  readonly hand?: string;
}

export interface Lesson {
  readonly number: number;
  readonly title: string;
  readonly goal: string;
  readonly pages: readonly Page[];
  readonly exercises: readonly Exercise[];
}

type Fields = Record<string, unknown>;

function fields(value: unknown, where: string): Fields {
  if (typeof value !== 'object' || value === null || Array.isArray(value)) {
    throw new Error(`${where}: expected an object`);
  }
  return value as Fields;
}

function text(source: Fields, key: string, where: string): string {
  const value = source[key];
  if (typeof value !== 'string' || value.trim() === '') {
    throw new Error(`${where}: "${key}" must be a non-empty text`);
  }
  return value;
}

function optionalText(source: Fields, key: string, where: string): string | undefined {
  return source[key] === undefined ? undefined : text(source, key, where);
}

function whole(source: Fields, key: string, where: string): number {
  const value = source[key];
  if (typeof value !== 'number' || !Number.isInteger(value)) {
    throw new Error(`${where}: "${key}" must be a whole number`);
  }
  return value;
}

function list(source: Fields, key: string, where: string): unknown[] {
  const value = source[key];
  if (!Array.isArray(value) || value.length === 0) {
    throw new Error(`${where}: "${key}" must be a non-empty list`);
  }
  return value;
}

function oneOf<T extends string>(source: Fields, key: string, allowed: readonly T[], where: string): T {
  const value = text(source, key, where);
  if (!allowed.includes(value as T)) {
    throw new Error(`${where}: "${key}" must be one of ${allowed.join(', ')}`);
  }
  return value as T;
}

function parseExercise(value: unknown, where: string): Exercise {
  const source = fields(value, where);
  const id = text(source, 'id', where);
  const here = `${where} (${id})`;
  const type = oneOf(source, 'type', ['choice', 'bid', 'play'] as const, here);
  if (type === 'choice') {
    const options = list(source, 'options', here).map((option) => {
      if (typeof option !== 'string') throw new Error(`${here}: every option must be a text`);
      return option;
    });
    const answer = whole(source, 'answer', here);
    if (answer < 0 || answer >= options.length) {
      throw new Error(`${here}: "answer" must point at one of the options`);
    }
    return {
      type,
      id,
      question: text(source, 'question', here),
      hand: optionalText(source, 'hand', here),
      options,
      answer,
      explanation: text(source, 'explanation', here),
    };
  }
  if (type === 'bid') {
    const auction = source.auction;
    if (typeof auction !== 'string') throw new Error(`${here}: "auction" must be a text`);
    return {
      type,
      id,
      hand: text(source, 'hand', here),
      dealer: oneOf(source, 'dealer', SEATS, here),
      auction,
    };
  }
  const hands = fields(source.hands, `${here} hands`);
  return {
    type,
    id,
    intro: text(source, 'intro', here),
    trump: oneOf(source, 'trump', STRAINS, here),
    leader: oneOf(source, 'leader', SEATS, here),
    hands: {
      N: text(hands, 'N', here),
      E: text(hands, 'E', here),
      S: text(hands, 'S', here),
      W: text(hands, 'W', here),
    },
    target: whole(source, 'target', here),
    explanation: text(source, 'explanation', here),
  };
}

// Checks the shape of a lesson file and gives it its types
export function parseLesson(data: unknown): Lesson {
  const source = fields(data, 'lesson');
  const number = whole(source, 'number', 'lesson');
  const where = `lesson ${number}`;
  return {
    number,
    title: text(source, 'title', where),
    goal: text(source, 'goal', where),
    pages: list(source, 'pages', where).map((page, index) => {
      const here = `${where} page ${index + 1}`;
      const pageFields = fields(page, here);
      return {
        title: text(pageFields, 'title', here),
        text: text(pageFields, 'text', here),
        hand: optionalText(pageFields, 'hand', here),
      };
    }),
    exercises: list(source, 'exercises', where).map((exercise, index) =>
      parseExercise(exercise, `${where} exercise ${index + 1}`),
    ),
  };
}
```

```ts
import {
  type Auction,
  type Call,
  type Card,
  type Hand,
  type PlayState,
  type Seat,
  auctionFrom,
  dummyOf,
  newAuction,
  parseCard,
  parseHand,
  sameCall,
  seatToCall,
  seatToPlay,
  seededRng,
  sideOf,
  tricksWon,
} from '../core';
import { type Choice, chooseCall } from '../bidding';
import { botCard, viewOfPlay } from '../bots';
import { solve } from '../solver';
import { type BidExercise, type PlayExercise } from './types';

// In a play exercise South is declarer and North is dummy
const DECLARER: Seat = 'S';
const NO_AUCTION: Auction = newAuction('N');
const DEFENDER_SAMPLES = 30;

export interface BidSetup {
  readonly hand: Hand;
  readonly auction: Auction;
  // The seat of the student
  readonly seat: Seat;
}

export function bidSetup(exercise: BidExercise): BidSetup {
  const auction = auctionFrom(exercise.dealer, exercise.auction);
  return { hand: parseHand(exercise.hand), auction, seat: seatToCall(auction) };
}

// The right answer: what the rules up to and including this lesson prescribe
export function bidAnswer(exercise: BidExercise, lessonNumber: number): Choice {
  const { hand, auction } = bidSetup(exercise);
  return chooseCall(hand, auction, { maxLesson: lessonNumber });
}

export function isRightCall(exercise: BidExercise, lessonNumber: number, call: Call): boolean {
  return sameCall(bidAnswer(exercise, lessonNumber).call, call);
}

function cards(text: string): Card[] {
  return text.trim().split(/\s+/).map(parseCard);
}

export function startPosition(exercise: PlayExercise): PlayState {
  return {
    trump: exercise.trump,
    declarer: DECLARER,
    hands: {
      N: cards(exercise.hands.N),
      E: cards(exercise.hands.E),
      S: cards(exercise.hands.S),
      W: cards(exercise.hands.W),
    },
    completed: [],
    current: { leader: exercise.leader, cards: [] },
  };
}

// True when the student chooses the next card: South plays both the North and the South hand
export function studentToPlay(state: PlayState): boolean {
  return sideOf(seatToPlay(state)) === sideOf(DECLARER);
}

export function tricksForStudent(state: PlayState): number {
  return tricksWon(state, sideOf(DECLARER));
}

// The most tricks North-South can take against perfect defence
export function bestPossible(exercise: PlayExercise): number {
  const state = startPosition(exercise);
  const results = solve(state).map((result) => result.tricks);
  const size = state.hands[seatToPlay(state)].length;
  return studentToPlay(state) ? Math.max(...results) : size - Math.max(...results);
}

// The card a strong defender plays; the positions are small, so this is quick
export function defenderCard(state: PlayState, seed: number): Card {
  const view = viewOfPlay(state, NO_AUCTION, true);
  return botCard(view, 'hard', seededRng(seed), { samples: DEFENDER_SAMPLES });
}

export function dummySeat(state: PlayState): Seat {
  return dummyOf(state);
}
```

`src/lessons/progress.ts`:

```ts
// What the student has finished and which exercises still need practice

export interface Trouble {
  readonly id: string;
  // Times in a row this exercise was answered right in practice
  readonly streak: number;
}

export interface Progress {
  // Numbers of the finished lessons
  readonly completed: readonly number[];
  readonly trouble: readonly Trouble[];
}

export const NO_PROGRESS: Progress = { completed: [], trouble: [] };

// Share of exercises that must be right at the first try to finish a lesson
export const PASS_SHARE = 0.8;
// An exercise leaves the practice list after this many right answers in a row
export const STREAK_TO_CLEAR = 2;

export function lessonPassed(firstTries: readonly boolean[]): boolean {
  if (firstTries.length === 0) return false;
  const right = firstTries.filter(Boolean).length;
  return right / firstTries.length >= PASS_SHARE;
}

export function highestCompleted(progress: Progress): number {
  return progress.completed.reduce((highest, number) => Math.max(highest, number), 0);
}

// A lesson is open when it is the first one, or when the lesson before it is finished
export function isOpen(progress: Progress, lessonNumber: number, unlockAll = false): boolean {
  return unlockAll || lessonNumber === 1 || progress.completed.includes(lessonNumber - 1);
}

// After a lesson: mark it finished when passed, and remember the exercises that went wrong
export function afterLesson(
  progress: Progress,
  lessonNumber: number,
  firstTries: Readonly<Record<string, boolean>>,
): Progress {
  const passed = lessonPassed(Object.values(firstTries));
  const completed =
    passed && !progress.completed.includes(lessonNumber)
      ? [...progress.completed, lessonNumber].sort((a, b) => a - b)
      : progress.completed;
  const wrong = Object.keys(firstTries).filter((id) => !firstTries[id]);
  const known = new Set(progress.trouble.map((item) => item.id));
  const trouble = [
    ...progress.trouble.map((item) => (wrong.includes(item.id) ? { ...item, streak: 0 } : item)),
    ...wrong.filter((id) => !known.has(id)).map((id) => ({ id, streak: 0 })),
  ];
  return { completed, trouble };
}

// After practising one exercise from the list
export function afterPractice(progress: Progress, id: string, right: boolean): Progress {
  const trouble = progress.trouble
    .map((item) => (item.id === id ? { ...item, streak: right ? item.streak + 1 : 0 } : item))
    .filter((item) => item.streak < STREAK_TO_CLEAR);
  return { ...progress, trouble };
}
```

### Task 5: Eén doorloop van een les

**Files:**
- Create: `src/lessons/session.ts`, `src/lessons/index.ts`
- Test: `src/lessons/session.test.ts`

**Interfaces:**
- Consumes: `Exercise`, `Lesson` uit `./types`; in de test `LESSONS` uit `./content`
- Produces:
  - `interface Session { lessonNumber: number | null; title: string; pages; page: number; queue: readonly Exercise[]; firstTries: Readonly<Record<string, boolean>>; lessonOf: Readonly<Record<string, number>>; total: number }` (`lessonNumber` is `null` voor de oefenlijst)
  - `type Stage = 'pages' | 'exercises' | 'done'`
  - `lessonSession(lesson: Lesson): Session`
  - `practiceSession(lessons: readonly Lesson[], ids: readonly string[]): Session`
  - `stageOf(session: Session): Stage`
  - `nextPage(session)`, `previousPage(session)`
  - `currentExercise(session: Session): Exercise | null`
  - `answered(session: Session, right: boolean): Session` (een fout antwoord zet de oefening achteraan)
  - `rightAtFirstTry(session: Session): number`
  - `src/lessons/index.ts` exporteert alles uit `types`, `exercise`, `progress`, `session`, plus `LESSONS` en `lessonByNumber`

```ts
import { describe, expect, it } from 'vitest';
import { LESSONS } from './content';
import {
  answered,
  currentExercise,
  lessonSession,
  nextPage,
  practiceSession,
  previousPage,
  rightAtFirstTry,
  stageOf,
} from './session';

const lesson = LESSONS[0];

function throughPages() {
  let session = lessonSession(lesson);
  while (stageOf(session) === 'pages') session = nextPage(session);
  return session;
}

describe('a lesson session', () => {
  it('starts with the pages', () => {
    const session = lessonSession(lesson);
    expect(session.title).toBe('Les 1: Bridge in vogelvlucht');
    expect(stageOf(session)).toBe('pages');
    expect(currentExercise(session)).toBeNull();
    expect(previousPage(session)).toBe(session);
    expect(previousPage(nextPage(session)).page).toBe(0);
  });

  it('goes on to the exercises after the last page', () => {
    const session = throughPages();
    expect(stageOf(session)).toBe('exercises');
    expect(currentExercise(session)).toBe(lesson.exercises[0]);
    expect(nextPage(session)).toBe(session);
  });

  it('moves on after a right answer', () => {
    const session = answered(throughPages(), true);
    expect(currentExercise(session)).toBe(lesson.exercises[1]);
    expect(session.firstTries).toEqual({ [lesson.exercises[0].id]: true });
  });

  it('brings a wrongly answered exercise back at the end', () => {
    let session = answered(throughPages(), false);
    const first = lesson.exercises[0];
    expect(currentExercise(session)).toBe(lesson.exercises[1]);
    expect(session.queue.at(-1)).toBe(first);
    while (currentExercise(session) !== first) session = answered(session, true);
    session = answered(session, true);
```

```ts
import { type Exercise, type Lesson } from './types';

// One run through the pages and exercises of a lesson, or through the practice list
export interface Session {
  // The lesson being followed, or null for the practice list
  readonly lessonNumber: number | null;
  readonly title: string;
  readonly pages: Lesson['pages'];
  // Index of the page being read; equal to pages.length once the exercises have started
  readonly page: number;
  // Exercises still to answer right, the current one first
  readonly queue: readonly Exercise[];
  // Per exercise: whether the first answer was right
  readonly firstTries: Readonly<Record<string, boolean>>;
  // Lesson number of every exercise, needed to check a bid
  readonly lessonOf: Readonly<Record<string, number>>;
  readonly total: number;
}

export type Stage = 'pages' | 'exercises' | 'done';

export function lessonSession(lesson: Lesson): Session {
  return {
    lessonNumber: lesson.number,
    title: `Les ${lesson.number}: ${lesson.title}`,
    pages: lesson.pages,
    page: 0,
    queue: lesson.exercises,
    firstTries: {},
    lessonOf: Object.fromEntries(lesson.exercises.map((exercise) => [exercise.id, lesson.number])),
    total: lesson.exercises.length,
  };
}

// A session over the exercises with these ids, taken from all lessons
export function practiceSession(lessons: readonly Lesson[], ids: readonly string[]): Session {
  const found = lessons.flatMap((lesson) =>
    lesson.exercises
      .filter((exercise) => ids.includes(exercise.id))
      .map((exercise) => ({ exercise, lessonNumber: lesson.number })),
  );
  return {
    lessonNumber: null,
    title: 'Oefen wat ik lastig vond',
    pages: [],
    page: 0,
    queue: found.map((item) => item.exercise),
    firstTries: {},
    lessonOf: Object.fromEntries(found.map((item) => [item.exercise.id, item.lessonNumber])),
    total: found.length,
  };
}

export function stageOf(session: Session): Stage {
  if (session.page < session.pages.length) return 'pages';
  return session.queue.length > 0 ? 'exercises' : 'done';
}

export function nextPage(session: Session): Session {
  return stageOf(session) === 'pages' ? { ...session, page: session.page + 1 } : session;
}

export function previousPage(session: Session): Session {
  return session.page > 0 && stageOf(session) === 'pages'
    ? { ...session, page: session.page - 1 }
    : session;
}

export function currentExercise(session: Session): Exercise | null {
  return stageOf(session) === 'exercises' ? session.queue[0] : null;
}

// Records the answer to the current exercise. A wrong answer sends the exercise to the back of the
// queue, so it comes back later in the same session.
export function answered(session: Session, right: boolean): Session {
  const exercise = currentExercise(session);
  if (!exercise) return session;
  const firstTries =
    exercise.id in session.firstTries
      ? session.firstTries
      : { ...session.firstTries, [exercise.id]: right };
  const rest = session.queue.slice(1);
  return { ...session, firstTries, queue: right ? rest : [...rest, exercise] };
}

export function rightAtFirstTry(session: Session): number {
  return Object.values(session.firstTries).filter(Boolean).length;
}
```

```ts
import { type Auction, type Card, type Hand, type Seat, seatToCall, seededRng, sideOf } from '../core';
import { type Choice } from '../bidding';
import {
  type Level,
  type PlayerView,
  botCall,
  botCard,
  heuristicCard,
  viewOfPlay,
} from '../bots';
import { type Game, PARTNER, USER, botCallIn, botCardIn, newGame, phaseOf, turnOf } from './game';
import { type Settings, effectiveProgress } from './settings';

export type BotRequest =
  | {
      readonly kind: 'call';
      readonly hand: Hand;
      readonly auction: Auction;
      readonly level: Level;
      readonly progressLesson: number;
    }
  | {
      readonly kind: 'card';
      readonly view: PlayerView;
      readonly level: Level;
      readonly seed: number;
      readonly budgetMs: number;
      readonly progressLesson: number;
    };

export type BotAnswer =
  | { readonly kind: 'call'; readonly choice: Choice }
  | { readonly kind: 'card'; readonly card: Card };

// Time a bot may spend simulating before it has to play
export const THINKING_BUDGET_MS = 1500;
const MAX_DEALS = 500;

// The partner of the user always plays at the hard level
export function levelFor(seat: Seat, settings: Settings): Level {
  return sideOf(seat) === sideOf(USER) ? 'hard' : settings.level;
}

// What to ask a bot now; null when it is the turn of the user or the game is over
export function botRequestFor(game: Game, settings: Settings): BotRequest | null {
  const turn = turnOf(game);
  if (!turn || turn.byUser) return null;
  const level = levelFor(turn.actor, settings);
  if (turn.kind === 'call') {
    return {
      kind: 'call',
      hand: game.deal[turn.seat],
      auction: game.auction,
      level,
      progressLesson: effectiveProgress(settings),
    };
  }
  if (!game.play) return null;
  const cardsPlayed = game.play.completed.length * 4 + game.play.current.cards.length;
  return {
    kind: 'card',
    view: viewOfPlay(game.play, game.auction),
    level,
    seed: game.seed * 100 + cardsPlayed,
    budgetMs: THINKING_BUDGET_MS,
    progressLesson: effectiveProgress(settings),
  };
}

export function answer(request: BotRequest): BotAnswer {
  if (request.kind === 'call') {
    const choice = botCall(request.hand, request.auction, request.level, request.progressLesson);
    return { kind: 'call', choice };
  }
  const card = botCard(request.view, request.level, seededRng(request.seed), {
    budgetMs: request.budgetMs,
    progressLesson: request.progressLesson,
  });
  return { kind: 'card', card };
}

// The same decision without simulation, for when the full answer takes too long
export function quickAnswer(request: BotRequest): BotAnswer {
  if (request.kind === 'call') return answer(request);
  return { kind: 'card', card: heuristicCard(request.view) };
}

export function applyAnswer(game: Game, botAnswer: BotAnswer): Game {
  return botAnswer.kind === 'call'
    ? botCallIn(game, botAnswer.choice)
    : botCardIn(game, botAnswer.card);
}

// Lets the bots bid all four hands
export function bidOutByBots(game: Game, settings: Settings): Game {
  let current = game;
  while (phaseOf(current) === 'auction') {
    const seat = seatToCall(current.auction);
    const level = seat === USER || seat === PARTNER ? 'hard' : settings.level;
    const choice = botCall(current.deal[seat], current.auction, level, effectiveProgress(settings));
    current = botCallIn(current, choice);
  }
  return current;
}

// For playing without bidding: the first deal from `firstSeed` on which the user becomes declarer
export function newDeclarerGame(board: number, firstSeed: number, settings: Settings): Game {
  for (let seed = firstSeed; seed < firstSeed + MAX_DEALS; seed++) {
    const game = bidOutByBots(newGame(board, seed), settings);
    if (game.contract?.declarer === USER) return game;
  }
  throw new Error('No deal found with the user as declarer');
}
```

```ts
const VERSION = 1;

  readonly progress: Progress;

  // True when stored data was unreadable and has been replaced by a fresh start

  game: null,
  nextBoard: 1,

type Store = Pick<Storage, 'getItem' | 'setItem'>;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null;
}

function usableProgress(value: unknown): Progress {
  if (!isRecord(value) || !Array.isArray(value.completed) || !Array.isArray(value.trouble)) {
    return NO_PROGRESS;
  }
  return {
    completed: value.completed.filter((item): item is number => typeof item === 'number'),
    trouble: value.trouble.filter(
      (item): item is Progress['trouble'][number] =>
        isRecord(item) && typeof item.id === 'string' && typeof item.streak === 'number',
    ),
  };
}

function usableGame(value: unknown): Game | null {
  if (!isRecord(value)) return null;
  try {
    phaseOf(value as unknown as Game);
    return value as unknown as Game;
  } catch {
    return null;
  }
}

export function load(store: Store): Loaded {
  const text = store.getItem(KEY);
  if (text === null) return { saved: FRESH, recovered: false };
  try {
    const data: unknown = JSON.parse(text);
    if (!isRecord(data) || data.version !== VERSION || !isRecord(data.settings)) {
      return { saved: FRESH, recovered: true };
    }
    const settings = { ...DEFAULT_SETTINGS, ...data.settings } as Settings;
    const nextBoard = typeof data.nextBoard === 'number' && data.nextBoard >= 1 ? data.nextBoard : 1;
    const game = data.game === null ? null : usableGame(data.game);
    const progress = usableProgress(data.progress);
    // The settings, the board number and the lesson progress survive a damaged game
    return {
      saved: { settings, game, nextBoard, progress },
      recovered: data.game !== null && game === null,
    };
  } catch {
    return { saved: FRESH, recovered: true };
  }
}

export function save(store: Store, saved: Saved): void {
  store.setItem(KEY, JSON.stringify({ version: VERSION, ...saved }));
}
```

```ts
  const turn = turnOf(game);
  if (!turn || turn.kind !== 'call' || !turn.byUser) return null;

export function cardHintFor(view: PlayerView, options: CardHintOptions): CardHint {
  const legal = legalCardsFor(view);
  if (legal.length === 1) {
    return { card: legal[0], text: 'Dit is de enige kaart die u mag spelen.' };
  }
  const thumb = heuristicChoice(view);
  const samples = options.samples ?? HINT_SAMPLES;
  if (samples === 0 || handToPlay(view).length > SIMULATION_MAX_CARDS) {
    return { card: thumb.card, text: RULE_TEXT[thumb.rule] };
  }

  const evaluations = evaluateCards(view, {
    samples,
    rng: seededRng(options.seed),
    limits: limitsFor(view, options.maxLesson),
    budgetMs: options.budgetMs,
    now: options.now,
  });
  const best = Math.max(...evaluations.map((evaluation) => evaluation.average));
  const bestCards = evaluations.filter((e) => best - e.average < EPSILON).map((e) => e.card);
  if (bestCards.length === evaluations.length) {
    return {
      card: thumb.card,
      text: `Alle kaarten leveren hier evenveel slagen op. ${RULE_TEXT[thumb.rule]}`,
    };
  }
  const card =
    bestCards.find((candidate) => sameCard(candidate, thumb.card)) ??
    bestCards.reduce((lowest, candidate) => (candidate.rank < lowest.rank ? candidate : lowest));
  const second = Math.max(
    ...evaluations.filter((e) => best - e.average >= EPSILON).map((e) => e.average),
  );
  return {
    card,
    text: `Deze kaart levert gemiddeld de meeste slagen op: ${tricksText(best)}. De beste andere kaart haalt ${tricksText(second)}.`,
  };
}

// The card to advise the user now; null when it is not the user's turn to play
export function cardHint(
  game: Game,
  settings: Settings,
  options: Partial<CardHintOptions> = {},
): CardHint | null {
  const turn = turnOf(game);
  if (!turn || turn.kind !== 'card' || !turn.byUser || !game.play) return null;
  const cardsPlayed = game.play.completed.length * 4 + game.play.current.cards.length;
  return cardHintFor(viewOfPlay(game.play, game.auction), {
    seed: game.seed * 100 + cardsPlayed + 50,
    maxLesson: lessonFor(settings),
    ...options,
  });
}

// What the call at this position means: the explanation the bot gave, or what the rules say
export function callMeaning(game: Game, index: number, settings: Settings): string {
  const note = game.notes[index];
  if (note) return note;
  const meanings = meaningsOf(game.auction, index, { maxLesson: lessonFor(settings) });
  return meanings.length > 0 ? meanings.join(' Of: ') : NO_AGREEMENT;
}
```

```ts
import { type Call, type Card } from '../core';
import {
  type BotAnswer,
  FRESH,
  type Game,
  type Saved,
  type Settings,
  applyAnswer,
  effectiveProgress,
  newDeclarerGame,
  newGame,
  undoMove,
  userCall,
  userCard,
} from '../game';
import {
  LESSONS,
  type Session,
  afterLesson,
  afterPractice,
  answered,
  currentExercise,
  highestCompleted,
  isOpen,
  lessonByNumber,
  lessonSession,
  nextPage,
  practiceSession,
  previousPage,
  stageOf,
} from '../lessons';

export type Screen = 'home' | 'table' | 'settings' | 'lessons' | 'lesson';

export interface AppState {
  readonly saved: Saved;
  readonly screen: Screen;
  // A message shown once on the home screen, or null
  readonly notice: string | null;
  // The lesson or practice round in progress; it is not saved
  readonly session: Session | null;
}

export type Action =
  | { readonly type: 'go'; readonly screen: Screen }
  | { readonly type: 'newBoard'; readonly seed: number }
  | { readonly type: 'replay' }
  | { readonly type: 'call'; readonly call: Call }
  | { readonly type: 'card'; readonly card: Card }
  | { readonly type: 'bot'; readonly answer: BotAnswer; readonly forGame: Game }
  | { readonly type: 'undo' }
  | { readonly type: 'settings'; readonly settings: Settings }
  | { readonly type: 'reset' }
  | { readonly type: 'dismissNotice' }
  | { readonly type: 'openLesson'; readonly number: number }
  | { readonly type: 'practice' }
  | { readonly type: 'lessonNext' }
  | { readonly type: 'lessonBack' }
  | { readonly type: 'lessonAnswer'; readonly right: boolean }
  | { readonly type: 'lessonClose' };

export const RECOVERED_NOTICE =
  'De opgeslagen gegevens konden niet worden gelezen. Het lopende spel is gewist.';
// Bidding is taught in lesson 4; before that the bots bid and the user only plays
const FIRST_BIDDING_LESSON = 4;

export function initialState(saved: Saved, recovered: boolean): AppState {
  return { saved, screen: 'home', notice: recovered ? RECOVERED_NOTICE : null, session: null };
}

// Whether the user bids in free play: the setting, and only once bidding has been taught
export function userBidsNow(settings: Settings): boolean {
  return settings.userBids && effectiveProgress(settings) >= FIRST_BIDDING_LESSON;
}

function deal(board: number, seed: number, settings: Settings): Game {
  return userBidsNow(settings) ? newGame(board, seed) : newDeclarerGame(board, seed, settings);
}

function withGame(state: AppState, game: Game | null): AppState {
  return { ...state, saved: { ...state.saved, game } };
}

function answerExercise(state: AppState, right: boolean): AppState {
  const { session, saved } = state;
  const exercise = session && currentExercise(session);
  if (!session || !exercise) return state;
  const next = answered(session, right);
  let progress = saved.progress;
  if (session.lessonNumber === null) {
    // Practice: every answer counts at once
    progress = afterPractice(progress, exercise.id, right);
  } else if (stageOf(next) === 'done') {
    progress = afterLesson(progress, session.lessonNumber, next.firstTries);
  }
  const settings = { ...saved.settings, progressLesson: highestCompleted(progress) };
  return { ...state, session: next, saved: { ...saved, progress, settings } };
}

export function reduce(state: AppState, action: Action): AppState {
  const { saved } = state;
  const { game } = saved;
  switch (action.type) {
    case 'go':
      return { ...state, screen: action.screen };
    case 'newBoard':
      return {
        ...state,
        screen: 'table',
        saved: {
          ...saved,
          game: deal(saved.nextBoard, action.seed, saved.settings),
          nextBoard: saved.nextBoard + 1,
        },
      };
    case 'replay':
      return game ? withGame(state, deal(game.board, game.seed, saved.settings)) : state;
    case 'call':
      return game ? withGame(state, userCall(game, action.call)) : state;
    case 'card':
      return game ? withGame(state, userCard(game, action.card)) : state;
    case 'bot':
      // An answer for an older position, for example after taking back a move, is dropped
      return game && game === action.forGame ? withGame(state, applyAnswer(game, action.answer)) : state;
    case 'undo':
      return game ? withGame(state, undoMove(game)) : state;
    case 'settings':
      return { ...state, saved: { ...saved, settings: action.settings } };
    case 'reset':
      return { saved: FRESH, screen: 'home', notice: null, session: null };
    case 'dismissNotice':
      return { ...state, notice: null };
    case 'openLesson': {
      const lesson = lessonByNumber(action.number);
      if (!lesson || !isOpen(saved.progress, lesson.number, saved.settings.unlockAll)) return state;
      return { ...state, screen: 'lesson', session: lessonSession(lesson) };
    }
    case 'practice': {
      const ids = saved.progress.trouble.map((item) => item.id);
      return ids.length === 0
        ? state
        : { ...state, screen: 'lesson', session: practiceSession(LESSONS, ids) };
    }
    case 'lessonNext':
      return state.session ? { ...state, session: nextPage(state.session) } : state;
    case 'lessonBack':
      return state.session ? { ...state, session: previousPage(state.session) } : state;
    case 'lessonAnswer':
      return answerExercise(state, action.right);
    case 'lessonClose':
      return { ...state, screen: 'lessons', session: null };
  }
}
```

```tsx
// Board 1 (North deals): South bids 1NT and everyone else passes
function southDeclares(): Game {
  let game = newGame(1, 1);
  while (phaseOf(game) === 'auction') {
    const turn = turnOf(game);
    game = turn?.byUser
      ? userCall(game, game.auction.calls.length === 2 ? bid(1, 'NT') : PASS)
      : botCallIn(game, pass);
  }
  return game;
}

function playCards(game: Game, count: number): Game {
  let current = game;
  for (let i = 0; i < count && current.play; i++) {
    const card = legalCards(current.play)[0];
    current = turnOf(current)?.byUser ? userCard(current, card) : botCardIn(current, card);
  }
  return current;
}
```

```tsx
import { useEffect, useState } from 'react';
import {
  type Call,
  type Card,
  type PlayState,
  SEATS,
  type Seat,
  isPlayComplete,
  legalCards,
  parseHand,
  playCard,
  sameCard,
  seatToPlay,
} from '../../core';
import { callLabel, meaningsOf } from '../../bidding';
import {
  type BidExercise,
  type ChoiceExercise,
  type Exercise,
  type PlayExercise,
  bidAnswer,
  bidSetup,
  defenderCard,
  isRightCall,
  startPosition,
  studentToPlay,
  tricksForStudent,
} from '../../lessons';
import { SEAT_NAME, cardLabel } from '../text';
import { AuctionGrid } from './AuctionGrid';
import { BiddingBox } from './BiddingBox';
import { HandView, HiddenHand } from './HandView';
import { TrickArea } from './TrickArea';

// Pause before a defender plays, so the student can follow the trick
const DEFENDER_DELAY_MS = 900;

interface Props<T extends Exercise> {
  readonly exercise: T;
  readonly lessonNumber: number;
  // Called when the student moves on, with whether the answer was right
  readonly onDone: (right: boolean) => void;
}

function Feedback({
  right,
  text,
  onDone,
}: {
  readonly right: boolean;
  readonly text: string;
  readonly onDone: () => void;
}) {
  return (
    <div className={right ? 'feedback feedback-right' : 'feedback feedback-wrong'} role="status">
      <p>
        <strong>{right ? 'Goed!' : 'Niet goed.'}</strong> {text}
      </p>
      <button type="button" className="primary" onClick={onDone}>
        Verder
      </button>
    </div>
  );
}

function ChoiceView({ exercise, onDone }: Props<ChoiceExercise>) {
  const [picked, setPicked] = useState<number | null>(null);
  return (
    <div className="exercise">
      <p className="question">{exercise.question}</p>
      {exercise.hand && <HandView hand={parseHand(exercise.hand)} />}
      <div className="options">
        {exercise.options.map((option, index) => (
          <button
            key={option}
            type="button"
            className={
              picked === null ? 'option-button' : index === exercise.answer ? 'option-button option-right' : 'option-button'
            }
            disabled={picked !== null}
            aria-pressed={picked === index}
            onClick={() => setPicked(index)}
          >
            {option}
          </button>
        ))}
      </div>
      {picked !== null && (
        <Feedback
          right={picked === exercise.answer}
          text={exercise.explanation}
          onDone={() => onDone(picked === exercise.answer)}
        />
      )}
    </div>
  );
}

function BidView({ exercise, lessonNumber, onDone }: Props<BidExercise>) {
  const [given, setGiven] = useState<Call | null>(null);
  const { hand, auction, seat } = bidSetup(exercise);
  const answer = bidAnswer(exercise, lessonNumber);
  const right = given !== null && isRightCall(exercise, lessonNumber, given);
  const meaning = (index: number) =>
    meaningsOf(auction, index, { maxLesson: lessonNumber }).join(' Of: ') || 'Voor dit bod is geen afspraak.';
  return (
    <div className="exercise">
      <p className="question">
        U zit {SEAT_NAME[seat]}. Wat biedt u met deze hand?
      </p>
      {auction.calls.length > 0 && (
        <AuctionGrid auction={auction} vulnerability="none" meaning={meaning} />
      )}
      <HandView hand={hand} />
      {given === null ? (
        <BiddingBox auction={auction} onCall={setGiven} />
      ) : (
        <Feedback
          right={right}
          text={right ? answer.explanation : `U bood ${callLabel(given)}. Het juiste bod is ${answer.explanation}`}
          onDone={() => onDone(right)}
        />
      )}
    </div>
  );
}

function PlayView({ exercise, onDone }: Props<PlayExercise>) {
  const [state, setState] = useState<PlayState>(() => startPosition(exercise));
  const [selected, setSelected] = useState<Card | null>(null);
  const over = isPlayComplete(state);
  const mine = !over && studentToPlay(state);
  const toPlay = over ? null : seatToPlay(state);
  const playable = mine ? legalCards(state) : [];
  const chosen = selected && playable.some((card) => sameCard(card, selected)) ? selected : null;
  const tricks = tricksForStudent(state);

  useEffect(() => {
    if (over || mine) return;
    const timer = setTimeout(() => {
      setState((current) =>
        isPlayComplete(current) || studentToPlay(current)
          ? current
          : playCard(current, defenderCard(current, current.completed.length * 4 + current.current.cards.length)),
      );
    }, DEFENDER_DELAY_MS);
    return () => clearTimeout(timer);
  }, [state, over, mine]);

  const playChosen = () => {
    if (chosen) {
      setState(playCard(state, chosen));
      setSelected(null);
    }
  };
  const select = (card: Card) => (chosen && sameCard(chosen, card) ? playChosen() : setSelected(card));

  const hand = (seat: Seat) => {
    if (seat === 'E' || seat === 'W') return <HiddenHand count={state.hands[seat].length} />;
    const active = seat === toPlay && mine;
    return (
      <HandView
        hand={state.hands[seat]}
        playable={active ? playable : undefined}
        selected={active ? chosen : null}
        onSelect={active ? select : undefined}
      />
    );
  };

  return (
    <div className="exercise">
      <p className="question">{exercise.intro}</p>
      <p className="play-count">
        Slagen voor u: {tricks} van de {exercise.target} die u nodig heeft.
      </p>
      <div className="table practice-table">
        {SEATS.map((seat) => (
          <section key={seat} className={`seat seat-${seat}`}>
            <h3 className={seat === toPlay ? 'seat-name to-play' : 'seat-name'}>
              {SEAT_NAME[seat]}
              {seat === 'N' ? ' (dummy)' : seat === 'S' ? ' (u)' : ''}
            </h3>
            {hand(seat)}
          </section>
        ))}
        <section className="centre">
          <TrickArea play={state} toPlay={toPlay} />
        </section>
      </div>
      {over ? (
        <Feedback
          right={tricks >= exercise.target}
          text={
            tricks >= exercise.target
              ? exercise.explanation
              : `U haalde ${tricks} van de ${exercise.target} slagen. ${exercise.explanation}`
          }
          onDone={() => onDone(tricks >= exercise.target)}
        />
      ) : (
        mine && (
          <button type="button" className="primary confirm" disabled={!chosen} onClick={playChosen}>
            {chosen ? `Speel ${cardLabel(chosen)}` : 'Kies een kaart'}
          </button>
        )
      )}
    </div>
  );
}

export function ExerciseView({ exercise, lessonNumber, onDone }: Props<Exercise>) {
  switch (exercise.type) {
    case 'choice':
      return <ChoiceView exercise={exercise} lessonNumber={lessonNumber} onDone={onDone} />;
    case 'bid':
      return <BidView exercise={exercise} lessonNumber={lessonNumber} onDone={onDone} />;
    case 'play':
      return <PlayView exercise={exercise} lessonNumber={lessonNumber} onDone={onDone} />;
  }
}
```

```tsx
import { parseHand } from '../../core';
import {
  type Session,
  currentExercise,
  lessonPassed,
  rightAtFirstTry,
  stageOf,
} from '../../lessons';
import { ExerciseView } from './ExerciseView';
import { HandView } from './HandView';

interface Props {
  readonly session: Session;
  readonly onNext: () => void;
  readonly onBack: () => void;
  readonly onAnswer: (right: boolean) => void;
  readonly onClose: () => void;
}

export function LessonScreen({ session, onNext, onBack, onAnswer, onClose }: Props) {
  const stage = stageOf(session);
  const exercise = currentExercise(session);
  const page = session.pages[session.page];
  // An exercise that comes back after a mistake starts fresh
  const attempt = session.total - session.queue.length + Object.keys(session.firstTries).length;

  return (
    <div className="lesson">
      <header className="lesson-head">
        <button type="button" onClick={onClose}>
          Stoppen
        </button>
        <h1>{session.title}</h1>
      </header>

      {stage === 'pages' && page && (
        <article className="lesson-page">
          <p className="lesson-step">
            Uitleg {session.page + 1} van {session.pages.length}
          </p>
          <h2>{page.title}</h2>
          {page.text.split('\n\n').map((paragraph) => (
            <p key={paragraph}>{paragraph}</p>
          ))}
          {page.hand && <HandView hand={parseHand(page.hand)} />}
          <div className="lesson-nav">
            <button type="button" disabled={session.page === 0} onClick={onBack}>
              Terug
            </button>
            <button type="button" className="primary" onClick={onNext}>
              {session.page + 1 === session.pages.length ? 'Naar de oefeningen' : 'Volgende'}
            </button>
          </div>
        </article>
      )}

      {stage === 'exercises' && exercise && (
        <section className="lesson-exercise">
          <p className="lesson-step">Nog {session.queue.length} te gaan</p>
          <ExerciseView
            key={`${exercise.id}-${attempt}`}
            exercise={exercise}
            lessonNumber={session.lessonOf[exercise.id]}
            onDone={onAnswer}
          />
        </section>
      )}

      {stage === 'done' && (
        <section className="lesson-done-panel">
          <h2>Klaar</h2>
          <p>
            U had {rightAtFirstTry(session)} van de {session.total} oefeningen in één keer goed.
          </p>
          {session.lessonNumber !== null &&
            (lessonPassed(Object.values(session.firstTries)) ? (
              <p>De les is afgerond. De volgende les is nu open.</p>
            ) : (
              <p>
                Om de les af te ronden moet u vier van de vijf oefeningen in één keer goed hebben.
                Lees de uitleg nog eens door en probeer het opnieuw.
              </p>
            ))}
          <button type="button" className="primary" onClick={onClose}>
            Terug naar het leerpad
          </button>
        </section>
      )}
    </div>
  );
}
```

`src/app/App.tsx`:

```tsx
import { useEffect, useReducer, useState } from 'react';
import { type Moment, callHint, callMeaning, cardName } from '../coach';
import { type Game, botRequestFor, load, phaseOf, save, turnOf } from '../game';
import { LESSONS } from '../lessons';
import { askBot, askHint, askReview } from './botClient';
import './coach.css';
import { HomeScreen } from './components/HomeScreen';
import { LessonScreen } from './components/LessonScreen';
import { LessonsScreen } from './components/LessonsScreen';
import './lessons.css';
import { SettingsScreen } from './components/SettingsScreen';
import { TableScreen } from './components/TableScreen';
import { initialState, reduce } from './state';

// Pause before a bot acts, so the user can follow the play; longer after a finished trick
const BOT_DELAY_MS = 700;
const TRICK_DELAY_MS = 1600;

function start() {
  const { saved, recovered } = load(window.localStorage);
  return initialState(saved, recovered);
}

export function App() {
  const [state, dispatch] = useReducer(reduce, undefined, start);
  const { saved, screen } = state;
  const { game, settings } = saved;

  useEffect(() => {
    save(window.localStorage, saved);
  }, [saved]);

  useEffect(() => {
    if (!game || screen !== 'table') return;
    const request = botRequestFor(game, settings);
    if (!request) return;
    let cancelled = false;
    const trickJustFinished =
      game.play !== null && game.play.current.cards.length === 0 && game.play.completed.length > 0;
    const timer = setTimeout(
      () => {
        void askBot(request).then((answer) => {
          if (!cancelled) dispatch({ type: 'bot', answer, forGame: game });
        });
      },
      trickJustFinished ? TRICK_DELAY_MS : BOT_DELAY_MS,
    );
    return () => {
      cancelled = true;
      clearTimeout(timer);
    };
  }, [game, settings, screen]);

  // Advice and review belong to one position; they are dropped as soon as the game moves on
  const [hint, setHint] = useState<{ game: Game; text: string } | null>(null);
  const [review, setReview] = useState<{ game: Game; moments: Moment[] } | null>(null);

  const showHint = () => {
    if (!game) return;
    const turn = turnOf(game);
    if (turn?.kind === 'call') {
      const advice = callHint(game, settings);
      if (advice) setHint({ game, text: `Advies: ${advice.explanation}` });
      return;
    }
    setHint({ game, text: 'Even denken…' });
    void askHint(game, settings).then((advice) => {
      if (advice) {
        setHint({ game, text: `Advies: speel ${cardName(advice.card)}. ${advice.text}` });
      }
    });
  };

  useEffect(() => {
    if (!game || screen !== 'table') return;
    const phase = phaseOf(game);
    if (phase !== 'finished' && phase !== 'passedOut') return;
    let cancelled = false;
    void askReview(game, settings).then((moments) => {
      if (!cancelled) setReview({ game, moments });
    });
    return () => {
      cancelled = true;
    };
  }, [game, settings, screen]);

  const newBoard = () => dispatch({ type: 'newBoard', seed: Math.floor(Math.random() * 1_000_000) + 1 });

  return (
    <div className={`app text-${settings.textSize}`}>
      {screen === 'table' && game ? (
        <TableScreen
          game={game}
          settings={settings}
          hint={hint?.game === game ? hint.text : null}
          review={review?.game === game ? review.moments : null}
          meaning={(index) => callMeaning(game, index, settings)}
          onHint={showHint}
          onCall={(call) => dispatch({ type: 'call', call })}
          onCard={(card) => dispatch({ type: 'card', card })}
          onUndo={() => dispatch({ type: 'undo' })}
          onNext={newBoard}
          onReplay={() => dispatch({ type: 'replay' })}
          onHome={() => dispatch({ type: 'go', screen: 'home' })}
        />
      ) : screen === 'lesson' && state.session ? (
        <LessonScreen
          session={state.session}
          onNext={() => dispatch({ type: 'lessonNext' })}
          onBack={() => dispatch({ type: 'lessonBack' })}
          onAnswer={(right) => dispatch({ type: 'lessonAnswer', right })}
          onClose={() => dispatch({ type: 'lessonClose' })}
        />
      ) : screen === 'lessons' ? (
        <LessonsScreen
          lessons={LESSONS}
          progress={saved.progress}
          unlockAll={settings.unlockAll}
          onOpen={(number) => dispatch({ type: 'openLesson', number })}
          onPractice={() => dispatch({ type: 'practice' })}
          onBack={() => dispatch({ type: 'go', screen: 'home' })}
        />
      ) : screen === 'settings' ? (
        <SettingsScreen
          settings={settings}
          onChange={(next) => dispatch({ type: 'settings', settings: next })}
          onReset={() => dispatch({ type: 'reset' })}
          onBack={() => dispatch({ type: 'go', screen: 'home' })}
        />
      ) : (
        <HomeScreen
          game={game}
          notice={state.notice}
          onContinue={() => dispatch({ type: 'go', screen: 'table' })}
          onNew={newBoard}
          onLessons={() => dispatch({ type: 'go', screen: 'lessons' })}
          onSettings={() => dispatch({ type: 'go', screen: 'settings' })}
          onDismissNotice={() => dispatch({ type: 'dismissNotice' })}
        />
      )}
    </div>
  );
}
```
