# App: de speelbare tafel — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Een app in de browser waarin de gebruiker een spel bridge biedt en speelt tegen de bots: speeltafel, biedbox, uitslag, instellingen, en een lopend spel dat bewaard blijft.

**Architecture:** De module `game` beschrijft één spel als onveranderlijke gegevens en bevat alle overgangen (een bod, een kaart, terugnemen), zonder schermcode. De module `app` is een dunne React-laag: één reducer voor de toestand van de app, een Web Worker waarin de bots rekenen, en schermen die de toestand tonen. Alles wat de gebruiker ziet is Nederlands.

**Tech Stack:** TypeScript (strict), React 19, Vite, Vitest. Bouwt op `src/core`, `src/bidding`, `src/bots`.

Spec: `docs/superpowers/specs/2026-10-03-bridge-app-design.md`, stap 5 van de bouwvolgorde. Vereist dat de plannen `2026-10-03-solver.md` en `2026-10-03-bots.md` zijn uitgevoerd.

## Wat deze stap wel en niet bevat

Wel: beginscherm, speeltafel met bieden en afspelen, uitslag met telling, instellingen (niveau, tekstgrootte, zelf bieden, punten tonen, alles wissen), opslag van het lopende spel, terugnemen van de laatste zet.

Niet, want dat hoort bij latere stappen: de hintknop en de nabespreking (stap 6), het leerpad (stap 7), installeren als app en offline werken (stap 8).

## Afspraken over het gedrag

- De gebruiker zit altijd Zuid; Noord is de partner.
- De partner speelt altijd op niveau `hard`; de tegenstanders op het niveau uit de instellingen.
- Is Zuid leider, dan speelt de gebruiker ook de kaarten van de dummy (Noord). Is Noord leider, dan speelt de bot beide handen en kijkt de gebruiker mee; de hand van Noord is dan zichtbaar.
- Een bod of kaart kiezen is aantikken en bevestigen. Een tweede tik op de gekozen kaart speelt hem ook.
- Kaarten die niet gespeeld mogen worden zijn grijs en niet aan te tikken; biedingen die niet mogen ook.
- Terugnemen kan tot het bieden is afgelopen, of tot de slag compleet is. De biedingen of kaarten van de bots na die zet verdwijnen mee.
- Na een bot-zet wacht de app 0,7 seconde; na een complete slag 1,6 seconde, zodat de slag te zien blijft.
- Een bot die na 5 seconden geen antwoord heeft gegeven, speelt op vuistregels.
- Met "Zelf bieden" uit bieden de bots alle vier de handen en wordt er gedeeld tot Zuid leider is.
- Tikken op een bod in het biedverloop toont de uitleg die de bot erbij gaf.
- "Nieuw spel" vraagt om bevestiging als er nog een spel loopt.
- Opslag gebeurt in `localStorage`. Is de opslag geblokkeerd of vol, dan werkt de app door zonder te bewaren. Het ontwerp noemde IndexedDB; voor deze kleine hoeveelheid gegevens is `localStorage` eenvoudiger en voldoende. Is het opgeslagen spel onleesbaar, dan blijven de instellingen behouden en krijgt de gebruiker een melding.

```ts
  PASS,
  SEATS,
  bid,
  callText,
  cardId,
  dummyOf,

  scoreContract,

  canUndo,
  handOf,
  newGame,
  phaseOf,
  resultOf,
  turnOf,
  undoMove,

const note = (text: string) => ({
  call: PASS,
  ruleId: 'test',
  explanation: text,
  source: 'rule' as const,
});

// Board 1: North deals. Everybody passes except the given seat, which bids the given contract.
function bidTo(game: Game, declarer: 'N' | 'S', contract = bid(1, 'NT')): Game {
  let current = game;
  while (phaseOf(current) === 'auction') {
    const turn = turnOf(current);
    const alreadyBid = current.auction.calls.some((call) => call.type === 'bid');
    const call = turn?.seat === declarer && !alreadyBid ? contract : PASS;
    current = turn?.byUser ? userCall(current, call) : botCallIn(current, { ...note('test'), call });
  }
  return current;
}

function playCards(game: Game, count: number): Game {
  let current = game;
  for (let i = 0; i < count; i++) {
    if (!current.play) throw new Error('not in play');
    const card = legalCards(current.play)[0];
    current = turnOf(current)?.byUser ? userCard(current, card) : botCardIn(current, card);
  }
  return current;
}

    const game = newGame(1, 1);

  it('keeps the explanation of every bot call', () => {
    let game = botCallIn(botCallIn(newGame(1, 1), note('eerste')), note('tweede'));

    expect(turnOf(afterLead)).toEqual({ kind: 'card', seat: 'S', actor: 'N', byUser: false });

  it('takes back a call, together with the bot calls after it', () => {

    const after = botCallIn(userCall(before, bid(1, 'H')), note('c'));
    expect(canUndo(before)).toBe(false);
    expect(canUndo(after)).toBe(true);
    expect(undoMove(after)).toEqual(before);

  it('cannot take back the call that ends the auction', () => {
    const game = bidTo(newGame(1, 1), 'N');

    expect(canUndo(undone)).toBe(false);
  });

  it('cannot take back a card once the trick is complete', () => {
    const game = playCards(bidTo(newGame(1, 1), 'S'), 4);
    expect(game.play?.completed).toHaveLength(1);
    expect(canUndo(game)).toBe(false);
    expect(undoMove(game)).toBe(game);
  });
});
```

```ts
import {
  type Auction,
  type Call,
  type Card,
  type Contract,
  type Deal,
  type Hand,
  type PlayState,
  SEATS,
  type Seat,
  type Vulnerability,
  addCall,
  contractOf,
  dealCards,
  dealerForBoard,
  dummyOf,
  isAuctionComplete,
  isPlayComplete,
  newAuction,
  playCard,
  scoreContract,
  seatToCall,
  seatToPlay,
  seededRng,
  sideOf,
  startPlay,
  tricksWon,
  vulnerabilityForBoard,
} from '../core';
import { type Choice } from '../bidding';

// The person at the tablet always sits South; North is the partner
export const USER: Seat = 'S';
export const PARTNER: Seat = 'N';

export interface Game {
  readonly board: number;
  readonly seed: number;
  readonly dealer: Seat;
  readonly vulnerability: Vulnerability;
  // The four hands as dealt
  readonly deal: Deal;
  readonly auction: Auction;
  // One entry per call: the explanation a bot gave, or null for a call by the user
  readonly notes: readonly (string | null)[];
  readonly contract: Contract | null;
  readonly play: PlayState | null;
  // The game before the last move of the user, while that move can still be taken back
  readonly undo: Game | null;
}

export type Phase = 'auction' | 'play' | 'passedOut' | 'finished';

export interface Turn {
  readonly kind: 'call' | 'card';
  // The seat that calls, or whose card is played
  readonly seat: Seat;
  // The player who decides: declarer also plays the cards of dummy
  readonly actor: Seat;
  readonly byUser: boolean;
}

export interface Result {
  readonly contract: Contract;
  readonly declarerTricks: number;
  // Tricks above (positive) or below (negative) the contract
  readonly difference: number;
  readonly declarerScore: number;
  // The same score seen from the side of the user
  readonly userScore: number;
}

export function newGame(board: number, seed: number): Game {
  const dealer = dealerForBoard(board);
  return {
    board,
    seed,
    dealer,
    vulnerability: vulnerabilityForBoard(board),
    deal: dealCards(seededRng(seed)),
    auction: newAuction(dealer),
    notes: [],
    contract: null,
    play: null,
    undo: null,
  };
}

export function phaseOf(game: Game): Phase {
  if (!isAuctionComplete(game.auction)) return 'auction';
  if (!game.play) return 'passedOut';
  return isPlayComplete(game.play) ? 'finished' : 'play';
}

export function turnOf(game: Game): Turn | null {
  const phase = phaseOf(game);
  if (phase === 'auction') {
    const seat = seatToCall(game.auction);
    return { kind: 'call', seat, actor: seat, byUser: seat === USER };
  }
  if (phase === 'play' && game.play) {
    const seat = seatToPlay(game.play);
    const actor = seat === dummyOf(game.play) ? game.play.declarer : seat;
    return { kind: 'card', seat, actor, byUser: actor === USER };
  }
  return null;
}

function addCallTo(game: Game, call: Call, note: string | null): Game {
  const auction = addCall(game.auction, call);
  const notes = [...game.notes, note];
  if (!isAuctionComplete(auction)) return { ...game, auction, notes };
  const contract = contractOf(auction);
  return {
    ...game,
    auction,
    notes,
    contract,
    play: contract ? startPlay(game.deal, contract) : null,
    undo: null,
  };
}

function addCardTo(game: Game, card: Card): Game {
  if (!game.play) {
    throw new Error('No play in progress');
  }
  const play = playCard(game.play, card);
  const trickFinished = play.completed.length > game.play.completed.length;
  return { ...game, play, undo: trickFinished ? null : game.undo };
}

// A move by the user can be taken back until the auction ends or the trick is complete
export function userCall(game: Game, call: Call): Game {
  const before = { ...game, undo: null };
  const after = addCallTo(before, call, null);
  return phaseOf(after) === 'auction' ? { ...after, undo: before } : after;
}

export function userCard(game: Game, card: Card): Game {
  const before = { ...game, undo: null };
  const after = addCardTo(before, card);
  const trickFinished = (after.play?.completed.length ?? 0) > (before.play?.completed.length ?? 0);
  return trickFinished ? after : { ...after, undo: before };
}

export function botCallIn(game: Game, choice: Choice): Game {
  return addCallTo(game, choice.call, choice.explanation);
}

export function botCardIn(game: Game, card: Card): Game {
  return addCardTo(game, card);
}

export function canUndo(game: Game): boolean {
  return game.undo !== null;
}

export function undoMove(game: Game): Game {
  return game.undo ?? game;
}

// The hands the user may look at
export function visibleSeats(game: Game): Seat[] {
  const phase = phaseOf(game);
  if (phase === 'finished' || phase === 'passedOut') return [...SEATS];
  if (phase === 'auction' || !game.play) return [USER];
  const dummy = dummyOf(game.play);
  const leadMade = game.play.completed.length > 0 || game.play.current.cards.length > 0;
  return SEATS.filter(
    (seat) =>
      seat === USER ||
      (seat === dummy && leadMade) ||
      // As dummy the user watches partner play the hand
      (seat === PARTNER && dummy === USER),
  );
}

// The cards a seat holds right now
export function handOf(game: Game, seat: Seat): Hand {
  return game.play ? game.play.hands[seat] : game.deal[seat];
}

export function resultOf(game: Game): Result | null {
  if (phaseOf(game) !== 'finished' || !game.play || !game.contract) return null;
  const contract = game.contract;
  const side = sideOf(contract.declarer);
  const declarerTricks = tricksWon(game.play, side);
  const declarerScore = scoreContract(contract, declarerTricks, game.vulnerability);
  return {
    contract,
    declarerTricks,
    difference: declarerTricks - (contract.level + 6),
    declarerScore,
    userScore: side === sideOf(USER) ? declarerScore : -declarerScore,
  };
}
```

```ts
import { PASS, bid, legalCards, sameCard } from '../core';
import {
  answer,
  applyAnswer,
  bidOutByBots,
  botRequestFor,
  levelFor,
  newDeclarerGame,
  quickAnswer,
} from './botTurn';
import { type Game, newGame, phaseOf, turnOf, userCall, userCard } from './game';
import { DEFAULT_SETTINGS, type Settings } from './settings';

const settings: Settings = { ...DEFAULT_SETTINGS, level: 'medium' };

// Plays a whole board: the bots answer, the user passes and plays the first legal card
function playBoard(start: Game): Game {
  let game = start;
  for (let step = 0; step < 400 && turnOf(game); step++) {
    const request = botRequestFor(game, settings);
    if (request) {

  }
```

```ts
import { type Auction, type Card, type Hand, type Seat, seatToCall, seededRng, sideOf } from '../core';
import { type Choice } from '../bidding';
import {
  type Level,
  type PlayerView,
  botCall,
  botCard,
  heuristicCard,
  viewOfPlay,
} from '../bots';
import { type Game, PARTNER, USER, botCallIn, botCardIn, newGame, phaseOf, turnOf } from './game';
import { type Settings } from './settings';

export type BotRequest =
  | {
      readonly kind: 'call';
      readonly hand: Hand;
      readonly auction: Auction;
      readonly level: Level;
      readonly progressLesson: number;
    }
  | {
      readonly kind: 'card';
      readonly view: PlayerView;
      readonly level: Level;
      readonly seed: number;
      readonly budgetMs: number;
      readonly progressLesson: number;
    };

export type BotAnswer =
  | { readonly kind: 'call'; readonly choice: Choice }
  | { readonly kind: 'card'; readonly card: Card };

// Time a bot may spend simulating before it has to play
export const THINKING_BUDGET_MS = 1500;
const MAX_DEALS = 500;

// The partner of the user always plays at the hard level
export function levelFor(seat: Seat, settings: Settings): Level {
  return sideOf(seat) === sideOf(USER) ? 'hard' : settings.level;
}

// What to ask a bot now; null when it is the turn of the user or the game is over
export function botRequestFor(game: Game, settings: Settings): BotRequest | null {
  const turn = turnOf(game);
  if (!turn || turn.byUser) return null;
  const level = levelFor(turn.actor, settings);
  if (turn.kind === 'call') {
    return {
      kind: 'call',
      hand: game.deal[turn.seat],
      auction: game.auction,
      level,
      progressLesson: settings.progressLesson,
    };
  }
  if (!game.play) return null;
  const cardsPlayed = game.play.completed.length * 4 + game.play.current.cards.length;
  return {
    kind: 'card',
    view: viewOfPlay(game.play, game.auction),
    level,
    seed: game.seed * 100 + cardsPlayed,
    budgetMs: THINKING_BUDGET_MS,
    progressLesson: settings.progressLesson,
  };
}

export function answer(request: BotRequest): BotAnswer {
  if (request.kind === 'call') {
    const choice = botCall(request.hand, request.auction, request.level, request.progressLesson);
    return { kind: 'call', choice };
  }
  const card = botCard(request.view, request.level, seededRng(request.seed), {
    budgetMs: request.budgetMs,
    progressLesson: request.progressLesson,
  });
  return { kind: 'card', card };
}

// The same decision without simulation, for when the full answer takes too long
export function quickAnswer(request: BotRequest): BotAnswer {
  if (request.kind === 'call') return answer(request);
  return { kind: 'card', card: heuristicCard(request.view) };
}

export function applyAnswer(game: Game, botAnswer: BotAnswer): Game {
  return botAnswer.kind === 'call'
    ? botCallIn(game, botAnswer.choice)
    : botCardIn(game, botAnswer.card);
}

// Lets the bots bid all four hands
export function bidOutByBots(game: Game, settings: Settings): Game {
  let current = game;
  while (phaseOf(current) === 'auction') {
    const seat = seatToCall(current.auction);
    const level = seat === USER || seat === PARTNER ? 'hard' : settings.level;
    const choice = botCall(current.deal[seat], current.auction, level, settings.progressLesson);
    current = botCallIn(current, choice);
  }
  return current;
}

// For playing without bidding: the first deal from `firstSeed` on which the user becomes declarer
export function newDeclarerGame(board: number, firstSeed: number, settings: Settings): Game {
  for (let seed = firstSeed; seed < firstSeed + MAX_DEALS; seed++) {
    const game = bidOutByBots(newGame(board, seed), settings);
    if (game.contract?.declarer === USER) return game;
  }
  throw new Error('No deal found with the user as declarer');
}
```

```ts
import { type Game, phaseOf } from './game';
import { DEFAULT_SETTINGS, type Settings } from './settings';

const KEY = 'bridge-app';
const VERSION = 1;

export interface Saved {
  readonly settings: Settings;
  readonly game: Game | null;
  // Number of the next board to deal
  readonly nextBoard: number;
}

export interface Loaded {
  readonly saved: Saved;
  // True when stored data was unreadable and has been replaced by a fresh start
  readonly recovered: boolean;
}

export const FRESH: Saved = { settings: DEFAULT_SETTINGS, game: null, nextBoard: 1 };

type Store = Pick<Storage, 'getItem' | 'setItem'>;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null;
}

function usableGame(value: unknown): Game | null {
  if (!isRecord(value) || !isRecord(value.deal) || !Array.isArray(value.notes)) return null;
  const deal = value.deal;
  if (!['N', 'E', 'S', 'W'].every((seat) => Array.isArray(deal[seat]))) return null;
  try {
    phaseOf(value as unknown as Game);
    return value as unknown as Game;
  } catch {
    return null;
  }
}

// Stored settings with a value this version does not know fall back to the default for that setting
function usableSettings(value: Record<string, unknown>): Settings {
  const merged = { ...DEFAULT_SETTINGS, ...value } as Settings;
  return {
    ...merged,
    level: ['easy', 'medium', 'hard'].includes(merged.level) ? merged.level : DEFAULT_SETTINGS.level,
    textSize: ['normal', 'large', 'extraLarge'].includes(merged.textSize)
      ? merged.textSize
      : DEFAULT_SETTINGS.textSize,
  };
}

export function load(store: Store): Loaded {
  const text = store.getItem(KEY);
  if (text === null) return { saved: FRESH, recovered: false };
  try {
    const data: unknown = JSON.parse(text);
    if (!isRecord(data) || data.version !== VERSION || !isRecord(data.settings)) {
      return { saved: FRESH, recovered: true };
    }
    const settings = usableSettings(data.settings);
    const nextBoard = typeof data.nextBoard === 'number' && data.nextBoard >= 1 ? data.nextBoard : 1;
    const game = data.game === null ? null : usableGame(data.game);
    // The settings and the board number survive a damaged game
    return { saved: { settings, game, nextBoard }, recovered: data.game !== null && game === null };
  } catch {
    return { saved: FRESH, recovered: true };
  }
}

export function save(store: Store, saved: Saved): void {
  store.setItem(KEY, JSON.stringify({ version: VERSION, ...saved }));
}
```

```ts
import { STRAIN_SYMBOL, callLabel } from '../bidding';
import { type Level } from '../bots';
import { type Result, type TextSize } from '../game';

export const SEAT_NAME: Readonly<Record<Seat, string>> = {
  N: 'Noord',
  E: 'Oost',
  S: 'Zuid',
  W: 'West',
};

export const LEVEL_NAME: Readonly<Record<Level, string>> = {
  easy: 'Makkelijk',
  medium: 'Gemiddeld',
  hard: 'Moeilijk',
};

export const TEXT_SIZE_NAME: Readonly<Record<TextSize, string>> = {
  normal: 'Normaal',
  large: 'Groot',
  extraLarge: 'Extra groot',
};

const VULNERABILITY_TEXT: Readonly<Record<Vulnerability, string>> = {
  none: 'Niemand kwetsbaar',
  NS: 'Noord-Zuid kwetsbaar',
  EW: 'Oost-West kwetsbaar',
  all: 'Iedereen kwetsbaar',
};

export function vulnerabilityText(vulnerability: Vulnerability): string {
  return VULNERABILITY_TEXT[vulnerability];
}

export function cardLabel(card: Card): string {
  return `${STRAIN_SYMBOL[card.suit]}${rankLabel(card.rank)}`;
}

// Short form for the table of calls: 'pas', 'X', 'XX', '1♥', '3SA'
export function shortCallLabel(call: Call): string {
  if (call.type === 'pass') return 'pas';
  if (call.type === 'double') return 'X';
  if (call.type === 'redouble') return 'XX';
  return callLabel(call);
}

export function contractText(contract: Contract): string {
  const doubling =
    contract.doubling === 'doubled'
      ? ' gedoubleerd'
      : contract.doubling === 'redoubled'
        ? ' geredoubleerd'
        : '';
  const label = callLabel(bid(contract.level, contract.strain));
  return `${label}${doubling} door ${SEAT_NAME[contract.declarer]}`;
}

export function resultText(result: Result): string {
  const { difference } = result;
  if (difference === 0) return 'Contract precies gemaakt';
  if (difference > 0) {
    return `Contract gemaakt met ${difference} ${difference === 1 ? 'overslag' : 'overslagen'}`;
  }
  return `${-difference} down`;
}

export function scoreText(result: Result): string {
  return result.userScore >= 0
    ? `${result.userScore} punten voor Noord-Zuid`
    : `${-result.userScore} punten voor Oost-West`;
}
```

```ts
import { type Call, type Card } from '../core';
import {
  type BotAnswer,
  FRESH,
  type Game,
  type Saved,
  type Settings,
  applyAnswer,
  newDeclarerGame,
  newGame,
  undoMove,
  userCall,
  userCard,
} from '../game';

export type Screen = 'home' | 'table' | 'settings';

export interface AppState {
  readonly saved: Saved;
  readonly screen: Screen;
  // A message shown once on the home screen, or null
  readonly notice: string | null;
}

export type Action =
  | { readonly type: 'go'; readonly screen: Screen }
  | { readonly type: 'newBoard'; readonly seed: number }
  | { readonly type: 'replay' }
  | { readonly type: 'call'; readonly call: Call }
  | { readonly type: 'card'; readonly card: Card }
  | { readonly type: 'bot'; readonly answer: BotAnswer; readonly forGame: Game }
  | { readonly type: 'undo' }
  | { readonly type: 'settings'; readonly settings: Settings }
  | { readonly type: 'reset' }
  | { readonly type: 'dismissNotice' };

export const RECOVERED_NOTICE =
  'De opgeslagen gegevens konden niet worden gelezen. Het lopende spel is gewist.';

export function initialState(saved: Saved, recovered: boolean): AppState {
  return { saved, screen: 'home', notice: recovered ? RECOVERED_NOTICE : null };
}

function deal(board: number, seed: number, settings: Settings): Game {
  return settings.userBids ? newGame(board, seed) : newDeclarerGame(board, seed, settings);
}

function withGame(state: AppState, game: Game | null): AppState {
  return { ...state, saved: { ...state.saved, game } };
}

export function reduce(state: AppState, action: Action): AppState {
  const { saved } = state;
  const { game } = saved;
  switch (action.type) {
    case 'go':
      return { ...state, screen: action.screen };
    case 'newBoard':
      return {
        ...state,
        screen: 'table',
        saved: {
          ...saved,
          game: deal(saved.nextBoard, action.seed, saved.settings),
          nextBoard: saved.nextBoard + 1,
        },
      };
    case 'replay':
      return game ? withGame(state, deal(game.board, game.seed, saved.settings)) : state;
    case 'call':
      return game ? withGame(state, userCall(game, action.call)) : state;
    case 'card':
      return game ? withGame(state, userCard(game, action.card)) : state;
    case 'bot':
      // An answer for an older position, for example after taking back a move, is dropped
      return game && game === action.forGame ? withGame(state, applyAnswer(game, action.answer)) : state;
    case 'undo':
      return game ? withGame(state, undoMove(game)) : state;
    case 'settings':
      return { ...state, saved: { ...saved, settings: action.settings } };
    case 'reset':
      return { saved: FRESH, screen: 'home', notice: null };
    case 'dismissNotice':
      return { ...state, notice: null };
  }
}
```

```ts
import { type BotAnswer, type BotRequest, quickAnswer } from '../game';

// A bot that has not answered after this long is replaced by rules of thumb
const TIMEOUT_MS = 5000;

let worker: Worker | null = null;
let nextId = 1;
const waiting = new Map<number, (answer: BotAnswer) => void>();

function botWorker(): Worker {
  if (!worker) {
    worker = new Worker(new URL('./botWorker.ts', import.meta.url), { type: 'module' });
    worker.onmessage = (event: MessageEvent<{ id: number; answer: BotAnswer }>) => {
      const resolve = waiting.get(event.data.id);
      waiting.delete(event.data.id);
      resolve?.(event.data.answer);
    };
  }
  return worker;
}

export function askBot(request: BotRequest): Promise<BotAnswer> {
  return new Promise((resolve) => {
    const id = nextId++;
    const fallback = () => {
      waiting.delete(id);
      resolve(quickAnswer(request));
    };
    const timer = setTimeout(fallback, TIMEOUT_MS);
    waiting.set(id, (answer) => {
      clearTimeout(timer);
      resolve(answer);
    });
    try {
      botWorker().postMessage({ id, request });
    } catch {
      clearTimeout(timer);
      fallback();
    }
  });
}
```

`src/app/App.tsx`:

```tsx
import { useEffect, useReducer } from 'react';
import { FRESH, botRequestFor, load, save, turnOf } from '../game';
import { askBot } from './botClient';
import { HomeScreen } from './components/HomeScreen';
import { SettingsScreen } from './components/SettingsScreen';
import { TableScreen } from './components/TableScreen';
import { initialState, reduce } from './state';

// Pause before a bot acts, so the user can follow the play; longer after a finished trick
const BOT_DELAY_MS = 700;
const TRICK_DELAY_MS = 1600;

// Storage can be blocked or full; the app then simply runs without saving
function start() {
  try {
    const { saved, recovered } = load(window.localStorage);
    return initialState(saved, recovered);
  } catch {
    return initialState(FRESH, false);
  }
}

export function App() {
  const [state, dispatch] = useReducer(reduce, undefined, start);
  const { saved, screen } = state;
  const { game, settings } = saved;

  useEffect(() => {
    try {
      save(window.localStorage, saved);
    } catch {
      // Nothing to do: the game goes on, it just is not kept
    }
  }, [saved]);

  useEffect(() => {
    if (!game || screen !== 'table') return;
    const request = botRequestFor(game, settings);
    if (!request) return;
    let cancelled = false;
    const trickJustFinished =
      game.play !== null && game.play.current.cards.length === 0 && game.play.completed.length > 0;
    const timer = setTimeout(
      () => {
        void askBot(request).then((answer) => {
          if (!cancelled) dispatch({ type: 'bot', answer, forGame: game });
        });
      },
      trickJustFinished ? TRICK_DELAY_MS : BOT_DELAY_MS,
    );
    return () => {
      cancelled = true;
      clearTimeout(timer);
    };
  }, [game, settings, screen]);

  const newBoard = () => dispatch({ type: 'newBoard', seed: Math.floor(Math.random() * 1_000_000) + 1 });

  return (
    <div className={`app text-${settings.textSize}`}>
      {screen === 'table' && game ? (
        <TableScreen
          game={game}
          settings={settings}
          onCall={(call) => dispatch({ type: 'call', call })}
          onCard={(card) => dispatch({ type: 'card', card })}
          onUndo={() => dispatch({ type: 'undo' })}
          onNext={newBoard}
          onReplay={() => dispatch({ type: 'replay' })}
          onHome={() => dispatch({ type: 'go', screen: 'home' })}
        />
      ) : screen === 'settings' ? (
        <SettingsScreen
          settings={settings}
          onChange={(next) => dispatch({ type: 'settings', settings: next })}
          onReset={() => dispatch({ type: 'reset' })}
          onBack={() => dispatch({ type: 'go', screen: 'home' })}
        />
      ) : (
        <HomeScreen
          game={game}
          notice={state.notice}
          onContinue={() => dispatch({ type: 'go', screen: 'table' })}
          onNew={() => {
            const running = game !== null && turnOf(game) !== null;
            if (!running || window.confirm('Het lopende spel stoppen en een nieuw spel beginnen?')) {
              newBoard();
            }
          }}
          onSettings={() => dispatch({ type: 'go', screen: 'settings' })}
          onDismissNotice={() => dispatch({ type: 'dismissNotice' })}
        />
      )}
    </div>
  );
}
```
