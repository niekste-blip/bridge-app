# Bots: bieden en afspelen op drie niveaus — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** De module `bots` bouwen: een bot die op drie niveaus een bod en een kaart kiest, zonder ooit in verborgen handen te kijken.

**Architecture:** Een bot krijgt nooit het hele spel, alleen een `PlayerView`: de handen die de speler mag zien, de gespeelde slagen en de overige kaarten zonder eigenaar. Bieden gaat via `chooseCall` uit `bidding`. Een kaart kiezen gaat op vuistregels (`heuristicCard`), of door de onbekende kaarten vele malen willekeurig te verdelen, elke verdeling met `solve` uit te rekenen en de kaart met het hoogste gemiddelde te nemen.

**Tech Stack:** TypeScript (strict), Vitest. Bouwt op `src/core`, `src/bidding` en `src/solver`.

Spec: `docs/superpowers/specs/2026-10-03-bridge-app-design.md`, stap 4 van de bouwvolgorde. Vereist dat het plan `2026-10-03-solver.md` is uitgevoerd.

## De drie niveaus

| Niveau | Bieden: regels tot en met les | Afspelen |
|---|---|---|
| `easy` | 7 | Vuistregels |
| `medium` | 10 | Simulatie met 20 verdelingen; kiest met 15 procent kans de op één na beste kaart |
| `hard` | 13 | Simulatie met 100 verdelingen; kiest altijd de beste kaart |

- De lesgrens voor het bieden is nooit lager dan 7 en nooit hoger dan de voortgang van de gebruiker (`lessonLimit`).
- Simulatie gebeurt alleen met 9 of minder kaarten in de hand. Met meer kaarten spelen alle niveaus op vuistregels, omdat de rekenmodule daar te traag voor is.
- De simulatie kan een tijdsbudget krijgen: na minstens vier verdelingen stopt ze zodra het budget op is. De app geeft later zo'n budget mee; tests niet.
- Bij gelijke gemiddelden wint de kaart die de vuistregels zouden spelen, anders de laagste.

Gemeten op 40 spellen met 12 verdelingen per beslissing: een leider met simulatie haalt gemiddeld een halve slag per spel meer dan een leider op vuistregels, en tegenspelers met simulatie houden de leider gemiddeld 0,3 slag per spel lager.

## Wat de simulatie weet

Een willekeurige verdeling van de onbekende kaarten klopt altijd met:

- de handen die de speler ziet en de kaarten die gespeeld zijn;
- het aantal kaarten dat elke verborgen speler nog heeft;
- kleuren die een speler niet meer bekende.

En waar mogelijk (binnen 40 pogingen) met wat de verborgen spelers in het bieden beloofden: het puntenbereik en de minimumlengtes uit `shownBy`. Lukt dat niet, bijvoorbeeld omdat iemand anders bood dan de regels voorschrijven, dan vervalt alleen die laatste eis.

    - `hands(north: string, east: string, south: string, west: string): Deal`

```ts
    expect(handToPlay(view)).toEqual(deal.N);

    const view = viewOfPlay(state, auction);
    expect(view.seat).toBe('E');
    expect(Object.keys(view.hands).sort()).toEqual(['E', 'N']);
    expect(hiddenSeats(view)).toEqual(['S', 'W']);
  });

  it('is the same whichever hidden hand holds a card', () => {
    let state = playCard(startPlay(deal, contract), deal.W[0]);
    state = playCard(state, legalCards(state)[0]);
    // East is to play; South and West are hidden. Swap one card between them.
    const south = [...state.hands.S];
    const west = [...state.hands.W];
    [south[0], west[0]] = [west[0], south[0]];
    const swapped = { ...state, hands: { ...state.hands, S: south, W: west } };

    let state = startPlay(deal, contract);
    for (let turn = 0; turn < 52; turn++) {
      const legal = legalCards(state);
      expect(legalCardsFor(viewOfPlay(state, auction)).map(cardId)).toEqual(legal.map(cardId));
      state = playCard(state, legal[turn % legal.length]);
```

```ts
import {
  type Auction,
  type Card,
  type CompletedTrick,
  type Hand,
  type PlayState,
  SEATS,
  type Seat,
  type Strain,
  type Trick,
  dummyOf,
  seatToPlay,
  sortHand,
} from '../core';

// Everything a player may know when choosing a card. Bots only ever receive this, never the full deal.
export interface PlayerView {
  // The seat whose card is played now
  readonly seat: Seat;
  // The player who chooses it: declarer also plays the cards of dummy
  readonly actor: Seat;
  readonly declarer: Seat;
  readonly dummy: Seat;
  readonly trump: Strain;
  readonly auction: Auction;
  // The hands the actor can see, always including the hand of `seat`
  readonly hands: Readonly<Partial<Record<Seat, Hand>>>;
  // The cards of the hidden hands together, without their owners
  readonly unseen: Hand;
  readonly completed: readonly CompletedTrick[];
  readonly current: Trick;
}

// `dummyOpen` shows dummy before any card is played, for practice positions that start mid-deal
export function viewOfPlay(state: PlayState, auction: Auction, dummyOpen = false): PlayerView {
  const seat = seatToPlay(state);
  const dummy = dummyOf(state);
  const actor = seat === dummy ? state.declarer : seat;
  // Dummy goes down after the opening lead
  const dummyVisible =
    dummyOpen || state.completed.length > 0 || state.current.cards.length > 0;

  const hands: Partial<Record<Seat, Hand>> = {};
  const unseen: Card[] = [];
  for (const other of SEATS) {
    if (other === actor || other === seat || (other === dummy && dummyVisible)) {
      hands[other] = state.hands[other];
    } else {
      unseen.push(...state.hands[other]);
    }
  }
  return {
    seat,
    actor,
    declarer: state.declarer,
    dummy,
    trump: state.trump,
    auction,
    hands,
    unseen: sortHand(unseen),
    completed: state.completed,
    current: state.current,
  };
}

export function handToPlay(view: PlayerView): Hand {
  return view.hands[view.seat] ?? [];
}

export function legalCardsFor(view: PlayerView): Card[] {
  const hand = handToPlay(view);
  if (view.current.cards.length === 0) return [...hand];
  const ledSuit = view.current.cards[0].suit;
  const following = hand.filter((card) => card.suit === ledSuit);
  return following.length > 0 ? following : [...hand];
}

export function hiddenSeats(view: PlayerView): Seat[] {
  return SEATS.filter((seat) => view.hands[seat] === undefined);
}
```

`src/bots/testing.ts`:

```ts
import {
  type Auction,
  type Card,
  type Deal,
  type PlayState,
  type Seat,
  type Strain,
  isPlayComplete,
  newAuction,
  parseCard,
  playCard,
} from '../core';
import { type PlayerView, viewOfPlay } from './view';

// Test helpers: small positions written as card ids, and playing a deal to the end

export function hands(north: string, east: string, south: string, west: string): Deal {
  const parse = (text: string) => (text === '' ? [] : text.split(' ').map(parseCard));
  return { N: parse(north), E: parse(east), S: parse(south), W: parse(west) };
}

// A position at the start of a trick, after one earlier trick so that dummy is on the table
export function position(deal: Deal, trump: Strain, declarer: Seat, leader: Seat): PlayState {
  return {
    trump,
    declarer,
    hands: deal,
    completed: [{ leader, cards: [], winner: leader }],
    current: { leader, cards: [] },
  };
}

export function play(state: PlayState, ...ids: string[]): PlayState {
  return ids.reduce((current, id) => playCard(current, parseCard(id)), state);
}

export function viewAt(state: PlayState, auction: Auction = newAuction('N')): PlayerView {
  return viewOfPlay(state, auction);
}

export function playOut(
  state: PlayState,
  auction: Auction,
  choose: (view: PlayerView) => Card,
): PlayState {
  let current = state;
  while (!isPlayComplete(current)) {
    current = playCard(current, choose(viewOfPlay(current, auction)));
  }
  return current;
}
```

- Consumes:
  - uit `'../core'`: `Card`, `SEATS`, `Suit`, `partnerOf(seat)`, `sideOf(seat)`, `trickWinner(trick, trump)`, `cardId(card)` (alleen in de test)
  - uit `./view`: `PlayerView`, `legalCardsFor(view)`
  - uit `./testing` (alleen in de test): `hands`, `play`, `position`, `viewAt`
- Produces:
  - `type HeuristicRule` (`'draw-trumps'`, `'cash-winner'`, `'long-suit'`, `'second-hand-low'`, `'third-hand-high'`, `'win-cheaply'`, `'play-low'`, `'ruff'`, `'discard'`)
  - `interface HeuristicChoice { readonly card: Card; readonly rule: HeuristicRule }`
  - `heuristicChoice(view: PlayerView): HeuristicChoice` (gooit een fout als er geen kaart te spelen is). De coach gebruikt `rule` later om een hint uit te leggen.
  - `heuristicCard(view: PlayerView): Card` (de kaart uit `heuristicChoice`)

Vuistregels bij uitkomen, in deze volgorde:

1. De leider (of de dummy) trekt troef zolang de tegenstanders nog troef kunnen hebben: de hoogste troef.
2. Een kaart die geen tegenstander kan slaan wordt geïncasseerd: hoger dan alle onbekende kaarten in die kleur en, voor een tegenspeler, hoger dan de kaarten van de dummy. De leider doet dit alleen als de tegenstanders geen troef meer hebben.
3. Anders de langste bijkleur: de hoogste van twee aansluitende honneurs (boer of hoger), anders de vierde van boven, anders de laagste.

Vuistregels bij bijspelen:

- Kleur bekennen: als tweede speler laag; als derde speler de hoogste kaart die de slag wint; als vierde de laagste kaart die de slag wint; laag als winnen niet kan.
- Ligt de slag al bij partner, dan laag, mits niemand meer na mij speelt, partner met een honneur wint of partner getroefd heeft.
- Niet kunnen bekennen: de laagste troef die de slag wint als een tegenstander aan slag is; anders de laagste kaart uit een bijkleur, bij gelijke hoogte uit de langste kleur.

```ts
import { type Card, SEATS, type Suit, partnerOf, sideOf, trickWinner } from '../core';
import { type PlayerView, legalCardsFor } from './view';

// The rule of thumb behind a card, so that a coach can explain it
export type HeuristicRule =
  | 'draw-trumps'
  | 'cash-winner'
  | 'long-suit'
  | 'second-hand-low'
  | 'third-hand-high'
  | 'win-cheaply'
  | 'play-low'
  | 'ruff'
  | 'discard';

export interface HeuristicChoice {
  readonly card: Card;
  readonly rule: HeuristicRule;
}

function lowest(cards: readonly Card[]): Card {
  return cards.reduce((best, card) => (card.rank < best.rank ? card : best));
}

function highest(cards: readonly Card[]): Card {
  return cards.reduce((best, card) => (card.rank > best.rank ? card : best));
}

function inSuit(cards: readonly Card[], suit: Suit): Card[] {
  return cards.filter((card) => card.suit === suit);
}

// Lowest card, from the longest suit when several are equally low
function discard(cards: readonly Card[]): Card {
  return cards.reduce((best, card) => {
    if (card.rank !== best.rank) return card.rank < best.rank ? card : best;
    return inSuit(cards, card.suit).length > inSuit(cards, best.suit).length ? card : best;
  });
}

function lead(view: PlayerView, legal: readonly Card[]): HeuristicChoice {
  const { trump, unseen } = view;
  const declaring = sideOf(view.seat) === sideOf(view.declarer);
  const unseenTrumps = trump === 'NT' ? [] : inSuit(unseen, trump);

  // Declarer draws trumps while the defenders may still have some
  if (declaring && trump !== 'NT' && unseenTrumps.length > 0) {
    const trumps = inSuit(legal, trump);
    if (trumps.length > 0) return { card: highest(trumps), rule: 'draw-trumps' };
  }

  // Cash a card that no opponent can beat
  const dummyCards = declaring ? [] : (view.hands[view.dummy] ?? []);
  const winners = legal.filter(
    (card) =>
      card.suit !== trump &&
      inSuit(unseen, card.suit).every((other) => other.rank < card.rank) &&
      inSuit(dummyCards, card.suit).every((other) => other.rank < card.rank) &&
      (!declaring || unseenTrumps.length === 0),
  );
  if (winners.length > 0) return { card: highest(winners), rule: 'cash-winner' };

  // Longest side suit: top of touching honours, else fourth best, else the lowest
  const sideCards = legal.filter((card) => card.suit !== trump);
  const pool = sideCards.length > 0 ? sideCards : legal;
  let suit = pool[0].suit;
  for (const card of pool) {
    if (inSuit(pool, card.suit).length > inSuit(pool, suit).length) suit = card.suit;
  }
  const cards = inSuit(pool, suit).sort((a, b) => b.rank - a.rank);
  if (cards.length >= 2 && cards[0].rank >= 11 && cards[0].rank - cards[1].rank === 1) {
    return { card: cards[0], rule: 'long-suit' };
  }
  return { card: cards.length >= 4 ? cards[3] : cards[cards.length - 1], rule: 'long-suit' };
}

function follow(view: PlayerView, legal: readonly Card[]): HeuristicChoice {
  const { current, trump } = view;
  const ledSuit = current.cards[0].suit;
  const winnerSeat = trickWinner(current, trump);
  const winnerIndex = (SEATS.indexOf(winnerSeat) - SEATS.indexOf(current.leader) + 4) % 4;
  const winning = current.cards[winnerIndex];
  const partnerWinning = winnerSeat === partnerOf(view.seat);

  const last = current.cards.length === 3;
  if (legal[0].suit === ledSuit) {
    const beating = winning.suit === ledSuit ? legal.filter((card) => card.rank > winning.rank) : [];
    // Partner's trick is safe when nobody plays after me, or when partner is winning with an honour
    const partnerHolds = partnerWinning && (last || winning.rank >= 11 || winning.suit !== ledSuit);
    if (beating.length === 0 || partnerHolds) return { card: lowest(legal), rule: 'play-low' };
    if (last) return { card: lowest(beating), rule: 'win-cheaply' };
    if (current.cards.length === 2) return { card: highest(beating), rule: 'third-hand-high' };
    return { card: lowest(legal), rule: 'second-hand-low' };
  }

  const trumps = trump === 'NT' ? [] : inSuit(legal, trump);
  if (!partnerWinning && trumps.length > 0) {
    const ruffs = winning.suit === trump ? trumps.filter((card) => card.rank > winning.rank) : trumps;
    if (ruffs.length > 0) return { card: lowest(ruffs), rule: 'ruff' };
  }
  const sideCards = legal.filter((card) => card.suit !== trump);
  return { card: discard(sideCards.length > 0 ? sideCards : legal), rule: 'discard' };
}

// The card a player following simple rules of thumb would play, with the rule behind it
export function heuristicChoice(view: PlayerView): HeuristicChoice {
  const legal = legalCardsFor(view);
  if (legal.length === 0) {
    throw new Error('No card to play');
  }
  return view.current.cards.length === 0 ? lead(view, legal) : follow(view, legal);
}

export function heuristicCard(view: PlayerView): Card {
  return heuristicChoice(view).card;
}
```

```ts
import {
  type Card,
  type Deal,
  type Hand,
  type Rng,
  SEATS,
  type Seat,
  type Suit,
  highCardPoints,
  seatAfter,
  suitLength,
} from '../core';
import { type Shown } from '../bidding';
import { type PlayerView, handToPlay, hiddenSeats } from './view';

export type Limits = Readonly<Partial<Record<Seat, Shown>>>;

function emptyRecord<T>(make: () => T): Record<Seat, T> {
  return { N: make(), E: make(), S: make(), W: make() };
}

// The cards every seat has played so far, taken from the tricks
export function cardsPlayedBy(view: PlayerView): Record<Seat, Card[]> {
  const played = emptyRecord<Card[]>(() => []);
  for (const trick of [...view.completed, view.current]) {
    trick.cards.forEach((card, index) => played[seatAfter(trick.leader, index)].push(card));
  }
  return played;
}

// Suits in which a seat failed to follow
export function knownVoids(view: PlayerView): Record<Seat, Set<Suit>> {
  const voids = emptyRecord<Set<Suit>>(() => new Set());
  for (const trick of [...view.completed, view.current]) {
    if (trick.cards.length === 0) continue;
    const ledSuit = trick.cards[0].suit;
    trick.cards.forEach((card, index) => {
      if (card.suit !== ledSuit) voids[seatAfter(trick.leader, index)].add(ledSuit);
    });
  }
  return voids;
}

export function shuffled<T>(items: readonly T[], rng: Rng): T[] {
  const result = [...items];
  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]];
  }
  return result;
}

// Deals the unseen cards over the hidden seats without breaking a known void; null when it gets stuck
function dealUnseen(
  view: PlayerView,
  hidden: readonly Seat[],
  voids: Record<Seat, Set<Suit>>,
  rng: Rng,
): Partial<Record<Seat, Card[]>> | null {
  const size = handToPlay(view).length;
  const inCurrentTrick = new Set(
    view.current.cards.map((_, index) => seatAfter(view.current.leader, index)),
  );
  const room = new Map<Seat, number>();
  const hands: Partial<Record<Seat, Card[]>> = {};
  for (const seat of hidden) {
    room.set(seat, size - (inCurrentTrick.has(seat) ? 1 : 0));
    hands[seat] = [];
  }
  const allowed = (card: Card) => hidden.filter((seat) => !voids[seat].has(card.suit));
  // Cards that can go to fewer seats are placed first
  const cards = shuffled(view.unseen, rng).sort((a, b) => allowed(a).length - allowed(b).length);
  for (const card of cards) {
    const options = allowed(card).filter((seat) => (room.get(seat) ?? 0) > 0);
    const total = options.reduce((sum, seat) => sum + (room.get(seat) ?? 0), 0);
    if (total === 0) return null;
    let pick = rng() * total;
    let chosen = options[options.length - 1];
    for (const seat of options) {
      pick -= room.get(seat) ?? 0;
      if (pick < 0) {
        chosen = seat;
        break;
      }
    }
    hands[chosen]?.push(card);
    room.set(chosen, (room.get(chosen) ?? 0) - 1);
  }
  return hands;
}

function withinLimits(hand: Hand, limits: Shown | undefined): boolean {
  if (!limits) return true;
  const points = highCardPoints(hand);
  if (points < limits.minHcp || points > limits.maxHcp) return false;
  return (['C', 'D', 'H', 'S'] as const).every(
    (suit) => suitLength(hand, suit) >= limits.minLength[suit],
  );
}

// A full deal that agrees with everything the actor knows: the visible hands, the cards played,
// the known voids and, when it can be done within `maxTries`, what the hidden players showed in the bidding
export function sampleDeal(view: PlayerView, rng: Rng, limits: Limits = {}, maxTries = 40): Deal {
  const hidden = hiddenSeats(view);
  const voids = knownVoids(view);
  const played = cardsPlayedBy(view);
  let fallback: Partial<Record<Seat, Card[]>> | null = null;
  let chosen: Partial<Record<Seat, Card[]>> | null = null;
  for (let attempt = 0; attempt < maxTries && !chosen; attempt++) {
    const hands = dealUnseen(view, hidden, voids, rng);
    if (!hands) continue;
    fallback = hands;
    const fits = hidden.every((seat) =>
      withinLimits([...(hands[seat] ?? []), ...played[seat]], limits[seat]),
    );
    if (fits) chosen = hands;
  }
  const hands = chosen ?? fallback ?? dealUnseen(view, hidden, emptyRecord(() => new Set()), rng) ?? {};
  const deal = emptyRecord<Hand>(() => []);
  for (const seat of SEATS) {
    deal[seat] = view.hands[seat] ?? hands[seat] ?? [];
  }
  return deal;
}
```

```ts
import { type Card, type PlayState, type Rng, cardId } from '../core';
import { solve } from '../solver';
import { type Limits, sampleDeal } from './sample';
import { type PlayerView } from './view';

export interface Evaluation {
  readonly card: Card;
  // Average number of tricks for the side to play over all sampled deals
  readonly average: number;
}

export interface SimulationOptions {
  readonly samples: number;
  readonly rng: Rng;
  readonly limits?: Limits;
  // Stop early when this much time has passed and at least `MIN_SAMPLES` deals were solved
  readonly budgetMs?: number;
  readonly now?: () => number;
}

const MIN_SAMPLES = 4;

// Deals the hidden cards at random many times and solves every deal with open cards
export function evaluateCards(view: PlayerView, options: SimulationOptions): Evaluation[] {
  const now = options.now ?? (() => performance.now());
  const start = now();
  const totals = new Map<string, { card: Card; sum: number }>();
  let solved = 0;
  for (let sample = 0; sample < options.samples; sample++) {
    if (
      options.budgetMs !== undefined &&
      solved >= MIN_SAMPLES &&
      now() - start > options.budgetMs
    ) {
      break;
    }
    const state: PlayState = {
      trump: view.trump,
      declarer: view.declarer,
      hands: sampleDeal(view, options.rng, options.limits),
      completed: view.completed,
      current: view.current,
    };
    for (const result of solve(state)) {
      const id = cardId(result.card);
      const entry = totals.get(id) ?? { card: result.card, sum: 0 };
      entry.sum += result.tricks;
      totals.set(id, entry);
    }
    solved++;
  }
  return [...totals.values()].map(({ card, sum }) => ({ card, average: sum / solved }));
}
```

```ts
export type Level = 'easy' | 'medium' | 'hard';

interface LevelSettings {
  // Highest lesson whose bidding rules the bot uses
  readonly lessonCap: number;
  // Deals to sample per card decision; 0 means rules of thumb only
  readonly samples: number;
  // Chance to play the second best card
  readonly slipChance: number;
}

const SETTINGS: Readonly<Record<Level, LevelSettings>> = {
  easy: { lessonCap: 7, samples: 0, slipChance: 0 },
  medium: { lessonCap: 10, samples: 20, slipChance: 0.15 },
  hard: { lessonCap: 13, samples: 100, slipChance: 0 },
};

// Opening, response and rebid belong together: a bot never knows less than lesson 7
export const BASE_LESSON = 7;
// With more cards than this in hand the search is too slow, and the bot plays by rules of thumb
export const SIMULATION_MAX_CARDS = 9;
const EPSILON = 1e-9;

export function lessonLimit(level: Level, progressLesson: number): number {
  return Math.max(BASE_LESSON, Math.min(SETTINGS[level].lessonCap, progressLesson));
}

// What the hidden players showed in the bidding, as limits for the simulation
export function limitsFor(view: PlayerView, maxLesson: number): Limits {
  return Object.fromEntries(
    hiddenSeats(view).map((seat) => [seat, shownBy(ALL_RULES, view.auction, seat, maxLesson)]),
  );
}

export function botCall(hand: Hand, auction: Auction, level: Level, progressLesson = 13): Choice {
  return chooseCall(hand, auction, { maxLesson: lessonLimit(level, progressLesson) });
}

export interface CardOptions {
  readonly samples?: number;
  readonly maxCards?: number;
  readonly budgetMs?: number;
  readonly now?: () => number;
  readonly progressLesson?: number;
}

export function botCard(view: PlayerView, level: Level, rng: Rng, options: CardOptions = {}): Card {
  const legal = legalCardsFor(view);
  if (legal.length === 1) return legal[0];
  const byRulesOfThumb = heuristicCard(view);
  const settings = SETTINGS[level];
  // A level that plays by rules of thumb stays that way, whatever the options say
  const samples = settings.samples === 0 ? 0 : (options.samples ?? settings.samples);
  if (samples === 0 || handToPlay(view).length > (options.maxCards ?? SIMULATION_MAX_CARDS)) {
    return byRulesOfThumb;
  }

  const evaluations = evaluateCards(view, {
    samples,
    rng,
    limits: limitsFor(view, lessonLimit(level, options.progressLesson ?? 13)),
    budgetMs: options.budgetMs,
    now: options.now,
  });

  const best = Math.max(...evaluations.map((evaluation) => evaluation.average));
  const isBest = (average: number) => best - average < EPSILON;
  if (rng() < settings.slipChance) {
    const rest = evaluations.filter((evaluation) => !isBest(evaluation.average));
    if (rest.length > 0) {
      const secondBest = Math.max(...rest.map((evaluation) => evaluation.average));
      return pick(rest.filter((e) => secondBest - e.average < EPSILON).map((e) => e.card), byRulesOfThumb);
    }
  }
  return pick(evaluations.filter((e) => isBest(e.average)).map((e) => e.card), byRulesOfThumb);
}

// Among equally good cards: the rule-of-thumb card when it is one of them, else the lowest
function pick(cards: readonly Card[], preferred: Card): Card {
  return (
    cards.find((card) => sameCard(card, preferred)) ??
    cards.reduce((lowest, card) => (card.rank < lowest.rank ? card : lowest))
  );
}
```

```ts
import { describe, expect, it } from 'vitest';
import {
  type Auction,
  type Contract,
  type Deal,
  SEATS,
  addCall,
  contractOf,
  dealCards,
  dealerForBoard,
  isAuctionComplete,
  isPlayComplete,
  newAuction,
  seatToCall,
  seededRng,
  sideOf,
  startPlay,
  tricksWon,
} from '../core';
import { type Level, botCall, botCard } from './index';
import { playOut } from './testing';

interface Board {
  readonly seed: number;
  readonly deal: Deal;
  readonly auction: Auction;
  readonly contract: Contract;
}

// The first `count` boards that are not passed out, bid by the bots
function boards(count: number): Board[] {
  const result: Board[] = [];
  for (let seed = 1; result.length < count; seed++) {
    const deal = dealCards(seededRng(seed));
    let auction = newAuction(dealerForBoard(seed));
    while (!isAuctionComplete(auction)) {
      auction = addCall(auction, botCall(deal[seatToCall(auction)], auction, 'hard').call);
    }
    const contract = contractOf(auction);
    if (contract) result.push({ seed, deal, auction, contract });
  }
  return result;
}

// Tricks for the declaring side when the two sides play at the given levels
function declarerTricks(board: Board, declaring: Level, defending: Level): number {
  const side = sideOf(board.contract.declarer);
  const rng = seededRng(board.seed * 7 + 1);
  const end = playOut(startPlay(board.deal, board.contract), board.auction, (view) =>
    botCard(view, sideOf(view.seat) === side ? declaring : defending, rng, {
      samples: 12,
      maxCards: 8,
    }),
  );
  return tricksWon(end, side);
}

describe('bots playing whole boards', () => {
  it('finish every board with legal cards only', () => {
    const levels: Level[] = ['easy', 'medium', 'hard'];
    for (const board of boards(12)) {
      const rng = seededRng(board.seed);
      const end = playOut(startPlay(board.deal, board.contract), board.auction, (view) =>
        botCard(view, levels[SEATS.indexOf(view.seat) % 3], rng, { samples: 6, maxCards: 6 }),
      );
      expect(isPlayComplete(end)).toBe(true);
      expect(end.completed).toHaveLength(13);
      expect(tricksWon(end, 'NS') + tricksWon(end, 'EW')).toBe(13);
    }
  });

  it('take more tricks with simulation than with rules of thumb', () => {
    let baseline = 0;
    let strongDeclarer = 0;
    let strongDefence = 0;
    for (const board of boards(16)) {
      baseline += declarerTricks(board, 'easy', 'easy');
      strongDeclarer += declarerTricks(board, 'hard', 'easy');
      strongDefence += declarerTricks(board, 'easy', 'hard');
    }
    expect(strongDeclarer).toBeGreaterThan(baseline);
    expect(strongDefence).toBeLessThan(baseline);
  }, 60000);
});
```
